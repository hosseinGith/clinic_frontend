import React, { useState } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import {
  Sparkles,
  Calendar,
  Clock,
  Stethoscope,
  CalendarDays,
  MessageSquare,
  AlertTriangle,
  X,
  Ban,
  FileText,
  RotateCw,
  ShieldCheck,
} from 'lucide-react';
import { services } from '../../../services';
import { platformQueryKeys } from '../../../services/queryKeys';
import type { Plan, BillingCycle } from '../../../types/platform/billing';
import { formatRialToToman } from '../../../utils/currency';
import { toPersianDigits } from '../../../utils/toPersianDigits';

interface ClinicBillingTabProps {
  clinicId: string;
}

export const ClinicBillingTab: React.FC<ClinicBillingTabProps> = ({ clinicId }) => {
  const queryClient = useQueryClient();

  // Modals state
  const [isChangePlanOpen, setIsChangePlanOpen] = useState(false);
  const [isExtendOpen, setIsExtendOpen] = useState(false);
  const [isExtendTrialOpen, setIsExtendTrialOpen] = useState(false);
  const [isCancelConfirmOpen, setIsCancelConfirmOpen] = useState(false);

  // Queries
  const { data: usageData, isLoading: isUsageLoading, error: usageError } = useQuery({
    queryKey: platformQueryKeys.clinicSubscription(clinicId),
    queryFn: () => services.platformBillingRepo.getClinicSubscription(clinicId),
  });

  const { data: plans = [] } = useQuery({
    queryKey: platformQueryKeys.plans(),
    queryFn: () => services.platformBillingRepo.getPlans(false),
  });

  const { data: invoicesData, isLoading: isInvoicesLoading } = useQuery({
    queryKey: platformQueryKeys.invoices({ clinicId }),
    queryFn: () => services.platformBillingRepo.getInvoices({ clinicId }),
  });

  // Mutations
  const changePlanMutation = useMutation({
    mutationFn: (dto: { planId: string; billingCycle: BillingCycle }) =>
      services.platformBillingRepo.changeClinicPlan(clinicId, dto),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: platformQueryKeys.clinicSubscription(clinicId) });
      queryClient.invalidateQueries({ queryKey: platformQueryKeys.invoices({ clinicId }) });
      setIsChangePlanOpen(false);
    },
  });

  const extendMutation = useMutation({
    mutationFn: (days: number) =>
      services.platformBillingRepo.extendClinicSubscription(clinicId, { days }),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: platformQueryKeys.clinicSubscription(clinicId) });
      setIsExtendOpen(false);
    },
  });

  const extendTrialMutation = useMutation({
    mutationFn: (days: number) =>
      services.platformBillingRepo.extendClinicTrial(clinicId, { days }),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: platformQueryKeys.clinicSubscription(clinicId) });
      setIsExtendTrialOpen(false);
    },
  });

  const cancelMutation = useMutation({
    mutationFn: () => services.platformBillingRepo.cancelClinicSubscription(clinicId),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: platformQueryKeys.clinicSubscription(clinicId) });
      setIsCancelConfirmOpen(false);
    },
  });

  if (isUsageLoading) {
    return (
      <div className="py-16 text-center">
        <div className="w-7 h-7 border-2 border-teal-500 border-t-transparent rounded-full animate-spin mx-auto mb-3" />
        <p className="text-xs text-slate-400">در حال بارگذاری وضعیت اشتراک...</p>
      </div>
    );
  }

  if (usageError || !usageData) {
    return (
      <div className="p-4 bg-rose-500/10 border border-rose-500/20 rounded-2xl text-rose-300 text-xs flex items-center gap-3">
        <AlertTriangle className="w-5 h-5 text-rose-400 shrink-0" strokeWidth={1.75} />
        <span>خطا در دریافت وضعیت اشتراک کلینیک: {(usageError as Error)?.message || 'اطلاعات در دسترس نیست'}</span>
      </div>
    );
  }

  const subscription = (usageData as any).subscription || (usageData as any).current || null;
  const plan = (usageData as any).plan || subscription?.plan || null;
  const daysRemaining = (usageData as any).daysRemaining ?? subscription?.daysRemaining ?? 0;

  const rawUsage = (usageData as any).usage;
  const usage = {
    doctorsCount: rawUsage?.doctorsCount ?? 0,
    maxDoctors: rawUsage?.maxDoctors ?? plan?.maxDoctors ?? 0,
    doctorsPercentage: rawUsage?.doctorsPercentage ?? 0,
    monthlyAppointmentsCount: rawUsage?.monthlyAppointmentsCount ?? 0,
    maxMonthlyAppointments: rawUsage?.maxMonthlyAppointments ?? plan?.maxMonthlyAppointments ?? 0,
    appointmentsPercentage: rawUsage?.appointmentsPercentage ?? 0,
    monthlySmsCount: rawUsage?.monthlySmsCount ?? 0,
    monthlySmsQuota: rawUsage?.monthlySmsQuota ?? plan?.monthlySmsQuota ?? 0,
    smsPercentage: rawUsage?.smsPercentage ?? 0,
  };

  const getProgressBarColor = (percentage: number) => {
    if (percentage >= 90) return 'from-rose-500 to-rose-400';
    if (percentage >= 75) return 'from-amber-500 to-amber-400';
    return 'from-teal-500 to-emerald-400';
  };

  return (
    <div className="space-y-6">
      {/* Current Subscription Card */}
      <div className="bg-slate-900/70 border border-slate-800/80 rounded-2xl p-6 shadow-xl shadow-black/20 backdrop-blur-xl space-y-6">
        <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4 border-b border-slate-800 pb-5">
          <div className="space-y-2">
            <div className="flex items-center gap-3 flex-wrap">
              <div className="p-2 rounded-xl bg-teal-500/10 text-teal-400 border border-teal-500/20">
                <Sparkles className="w-5 h-5" strokeWidth={1.75} />
              </div>
              <span className="text-xs text-slate-400 font-medium">پلن فعلی:</span>
              <span className="text-lg font-bold text-slate-100">{plan ? plan.name : 'بدون پلن فعال'}</span>
              {subscription && (
                <span
                  className={`inline-flex items-center gap-1.5 text-xs px-2.5 py-0.5 rounded-full border ${
                    subscription.status === 'active'
                      ? 'bg-emerald-500/10 text-emerald-400 border-emerald-500/20'
                      : subscription.status === 'trial'
                      ? 'bg-indigo-500/10 text-indigo-400 border-indigo-500/20'
                      : 'bg-rose-500/10 text-rose-400 border-rose-500/20'
                  }`}
                >
                  <span
                    className={`w-1.5 h-1.5 rounded-full ${
                      subscription.status === 'active'
                        ? 'bg-emerald-400 animate-pulse'
                        : subscription.status === 'trial'
                        ? 'bg-indigo-400'
                        : 'bg-rose-400'
                    }`}
                  />
                  {subscription.status === 'active' ? 'فعال' : subscription.status === 'trial' ? 'آزمایشی' : subscription.status}
                </span>
              )}
            </div>

            {subscription && (
              <div className="text-xs text-slate-400 flex items-center gap-3 flex-wrap">
                <span className="flex items-center gap-1">
                  <RotateCw className="w-3.5 h-3.5 text-slate-500" strokeWidth={1.75} />
                  دوره: <strong className="text-slate-200">{subscription.billingCycle === 'yearly' ? 'سالانه' : 'ماهانه'}</strong>
                </span>
                <span className="text-slate-600">•</span>
                <span className="flex items-center gap-1">
                  <Calendar className="w-3.5 h-3.5 text-slate-500" strokeWidth={1.75} />
                  پایان دوره: <strong className="text-slate-200 font-mono">{subscription.currentPeriodEnd ? subscription.currentPeriodEnd.slice(0, 10) : '—'}</strong>
                </span>
                <span className="text-slate-600">•</span>
                <span className="flex items-center gap-1">
                  <Clock className="w-3.5 h-3.5 text-slate-500" strokeWidth={1.75} />
                  باقیمانده: <strong className="text-teal-400 font-mono">{toPersianDigits(daysRemaining)} روز</strong>
                </span>
                {!subscription.autoRenew && (
                  <span className="inline-flex items-center gap-1 text-amber-400 font-medium px-2 py-0.5 rounded-md bg-amber-500/10 border border-amber-500/20">
                    <AlertTriangle className="w-3 h-3" strokeWidth={1.75} />
                    لغو شده در پایان دوره
                  </span>
                )}
              </div>
            )}
          </div>

          {/* Action Buttons */}
          <div className="flex flex-wrap items-center gap-2">
            <button
              onClick={() => setIsChangePlanOpen(true)}
              className="flex items-center gap-1.5 px-3.5 py-2 bg-teal-600 hover:bg-teal-500 text-white rounded-xl text-xs font-medium transition-all shadow-lg shadow-teal-600/20 active:scale-95"
            >
              <Sparkles className="w-3.5 h-3.5" strokeWidth={1.75} />
              تغییر پلن
            </button>
            <button
              onClick={() => setIsExtendOpen(true)}
              className="flex items-center gap-1.5 px-3.5 py-2 bg-slate-800/80 hover:bg-slate-700 text-slate-200 rounded-xl text-xs font-medium transition-colors border border-slate-700/80"
            >
              <Calendar className="w-3.5 h-3.5 text-slate-400" strokeWidth={1.75} />
              تمدید دوره
            </button>
            <button
              onClick={() => setIsExtendTrialOpen(true)}
              className="flex items-center gap-1.5 px-3.5 py-2 bg-indigo-500/10 hover:bg-indigo-500/20 text-indigo-300 rounded-xl text-xs font-medium transition-colors border border-indigo-500/30"
            >
              <Clock className="w-3.5 h-3.5 text-indigo-400" strokeWidth={1.75} />
              تمدید Trial
            </button>
            {subscription && subscription.autoRenew && (
              <button
                onClick={() => setIsCancelConfirmOpen(true)}
                className="flex items-center gap-1.5 px-3.5 py-2 bg-rose-500/10 hover:bg-rose-500/20 text-rose-300 rounded-xl text-xs font-medium transition-colors border border-rose-500/30"
              >
                <Ban className="w-3.5 h-3.5 text-rose-400" strokeWidth={1.75} />
                لغو اشتراک
              </button>
            )}
          </div>
        </div>

        {/* Usage vs Limits Progress Bars */}
        <div className="space-y-4">
          <div className="flex items-center justify-between">
            <h4 className="text-xs font-bold text-slate-300 flex items-center gap-2">
              <ShieldCheck className="w-4 h-4 text-teal-400" strokeWidth={1.75} />
              مصرف منابع نسبت به سقف پلن
            </h4>
          </div>

          <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
            {/* Doctors */}
            <div className="bg-slate-950/60 p-4 rounded-xl border border-slate-800/80 space-y-3">
              <div className="flex items-center justify-between text-xs">
                <span className="text-slate-400 flex items-center gap-1.5">
                  <Stethoscope className="w-3.5 h-3.5 text-teal-400" strokeWidth={1.75} />
                  پزشکان فعال:
                </span>
                <span className="font-mono font-bold text-slate-200">
                  {toPersianDigits(usage.doctorsCount)} / {usage.maxDoctors === 0 ? 'نامحدود' : toPersianDigits(usage.maxDoctors)}
                </span>
              </div>
              <div className="w-full bg-slate-800 rounded-full h-1.5 overflow-hidden">
                <div
                  className={`h-full rounded-full bg-gradient-to-r ${getProgressBarColor(usage.doctorsPercentage)} transition-all duration-500`}
                  style={{ width: `${Math.min(100, usage.doctorsPercentage)}%` }}
                />
              </div>
            </div>

            {/* Appointments */}
            <div className="bg-slate-950/60 p-4 rounded-xl border border-slate-800/80 space-y-3">
              <div className="flex items-center justify-between text-xs">
                <span className="text-slate-400 flex items-center gap-1.5">
                  <CalendarDays className="w-3.5 h-3.5 text-teal-400" strokeWidth={1.75} />
                  ویزیت‌های ماه جاری:
                </span>
                <span className="font-mono font-bold text-slate-200">
                  {toPersianDigits(usage.monthlyAppointmentsCount)} / {usage.maxMonthlyAppointments === 0 ? 'نامحدود' : toPersianDigits(usage.maxMonthlyAppointments)}
                </span>
              </div>
              <div className="w-full bg-slate-800 rounded-full h-1.5 overflow-hidden">
                <div
                  className={`h-full rounded-full bg-gradient-to-r ${getProgressBarColor(usage.appointmentsPercentage)} transition-all duration-500`}
                  style={{ width: `${Math.min(100, usage.appointmentsPercentage)}%` }}
                />
              </div>
            </div>

            {/* SMS */}
            <div className="bg-slate-950/60 p-4 rounded-xl border border-slate-800/80 space-y-3">
              <div className="flex items-center justify-between text-xs">
                <span className="text-slate-400 flex items-center gap-1.5">
                  <MessageSquare className="w-3.5 h-3.5 text-teal-400" strokeWidth={1.75} />
                  پیامک‌های ارسالی:
                </span>
                <span className="font-mono font-bold text-slate-200">
                  {toPersianDigits(usage.monthlySmsCount)} / {toPersianDigits(usage.monthlySmsQuota)}
                </span>
              </div>
              <div className="w-full bg-slate-800 rounded-full h-1.5 overflow-hidden">
                <div
                  className={`h-full rounded-full bg-gradient-to-r ${getProgressBarColor(usage.smsPercentage)} transition-all duration-500`}
                  style={{ width: `${Math.min(100, usage.smsPercentage)}%` }}
                />
              </div>
            </div>
          </div>
        </div>
      </div>

      {/* Invoices Table for this Clinic */}
      <div className="bg-slate-900/70 border border-slate-800/80 rounded-2xl overflow-hidden shadow-xl shadow-black/20 backdrop-blur-xl">
        <div className="p-4 border-b border-slate-800 flex items-center justify-between">
          <div className="flex items-center gap-2">
            <div className="p-1.5 rounded-lg bg-teal-500/10 text-teal-400 border border-teal-500/20">
              <FileText className="w-4 h-4" strokeWidth={1.75} />
            </div>
            <h3 className="text-sm font-bold text-slate-200">فاکتورها و سوابق مالی این کلینیک</h3>
          </div>
          <span className="text-xs text-slate-400">
            تعداد: <span className="font-mono font-bold text-slate-200">{toPersianDigits(invoicesData?.total || 0)}</span>
          </span>
        </div>

        {isInvoicesLoading ? (
          <div className="py-12 text-center text-slate-400 text-xs">
            <div className="w-6 h-6 border-2 border-teal-500 border-t-transparent rounded-full animate-spin mx-auto mb-2" />
            در حال بارگذاری فاکتورها...
          </div>
        ) : !invoicesData || invoicesData.items.length === 0 ? (
          <div className="py-12 text-center text-slate-400 text-xs">
            <FileText className="w-8 h-8 text-slate-600 mx-auto mb-2" strokeWidth={1.5} />
            فاکتوری برای این کلینیک ثبت نشده است.
          </div>
        ) : (
          <div className="overflow-x-auto">
            <table className="w-full text-right text-xs">
              <thead className="bg-slate-950/60 text-slate-400 uppercase text-[11px] font-semibold tracking-wider border-b border-slate-800">
                <tr>
                  <th className="py-3 px-4">شماره فاکتور</th>
                  <th className="py-3 px-4">مبلغ</th>
                  <th className="py-3 px-4">دوره</th>
                  <th className="py-3 px-4">وضعیت</th>
                  <th className="py-3 px-4">تاریخ پرداخت</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-800/60 text-slate-200">
                {invoicesData.items.map((inv) => (
                  <tr key={inv.id} className="hover:bg-slate-800/50 transition-colors">
                    <td className="py-3.5 px-4 font-mono text-teal-400">
                      <span className="px-2 py-0.5 rounded-md bg-slate-800 text-teal-300 border border-slate-700/60">
                        {inv.number}
                      </span>
                    </td>
                    <td className="py-3.5 px-4 font-bold font-mono">
                      {formatRialToToman(inv.amount)} <span className="text-slate-400 text-[11px] font-normal">تومان</span>
                    </td>
                    <td className="py-3.5 px-4 text-slate-300 font-mono text-[11px]">
                      {inv.periodStart.slice(0, 10)} تا {inv.periodEnd.slice(0, 10)}
                    </td>
                    <td className="py-3.5 px-4">
                      {inv.status === 'paid' && (
                        <span className="inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-full text-[11px] font-medium bg-emerald-500/10 text-emerald-400 border border-emerald-500/20">
                          <span className="w-1.5 h-1.5 rounded-full bg-emerald-400" />
                          پرداخت‌شده
                        </span>
                      )}
                      {inv.status === 'open' && (
                        <span className="inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-full text-[11px] font-medium bg-amber-500/10 text-amber-400 border border-amber-500/20">
                          <span className="w-1.5 h-1.5 rounded-full bg-amber-400" />
                          باز (پرداخت‌نشده)
                        </span>
                      )}
                      {inv.status === 'void' && (
                        <span className="inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-full text-[11px] font-medium bg-slate-800 text-slate-400 border border-slate-700">
                          <span className="w-1.5 h-1.5 rounded-full bg-slate-400" />
                          باطل‌شده
                        </span>
                      )}
                    </td>
                    <td className="py-3.5 px-4 text-slate-400 font-mono">
                      {inv.paidAt ? inv.paidAt.slice(0, 10) : '—'}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}
      </div>

      {/* Change Plan Modal */}
      {isChangePlanOpen && subscription && (
        <ChangePlanModal
          currentPlanId={subscription.planId}
          currentCycle={subscription.billingCycle}
          plans={plans}
          onClose={() => setIsChangePlanOpen(false)}
          onSubmit={async (planId, billingCycle) => {
            await changePlanMutation.mutateAsync({ planId, billingCycle });
          }}
          isSubmitting={changePlanMutation.isPending}
        />
      )}

      {/* Extend Modal */}
      {isExtendOpen && (
        <ExtendDaysModal
          title="تمدید دوره اشتراک"
          description="تعداد روزهای اضافه شده به پایان دوره فعلی این کلینیک را وارد کنید:"
          defaultDays={30}
          onClose={() => setIsExtendOpen(false)}
          onSubmit={async (days) => {
            await extendMutation.mutateAsync(days);
          }}
          isSubmitting={extendMutation.isPending}
        />
      )}

      {/* Extend Trial Modal */}
      {isExtendTrialOpen && (
        <ExtendDaysModal
          title="تمدید دوره آزمایشی (Trial)"
          description="تعداد روزهای تمدید مهلت تست رایگان برای این کلینیک:"
          defaultDays={14}
          onClose={() => setIsExtendTrialOpen(false)}
          onSubmit={async (days) => {
            await extendTrialMutation.mutateAsync(days);
          }}
          isSubmitting={extendTrialMutation.isPending}
        />
      )}

      {/* Cancel Confirm Modal */}
      {isCancelConfirmOpen && (
        <div className="fixed inset-0 z-50 bg-black/80 backdrop-blur-md flex items-center justify-center p-4 animate-in fade-in duration-200">
          <div className="bg-slate-900 border border-slate-800 rounded-2xl max-w-sm w-full p-6 space-y-4 text-xs shadow-2xl">
            <div className="flex items-center gap-3">
              <div className="p-2 rounded-xl bg-rose-500/10 text-rose-400 border border-rose-500/20">
                <AlertTriangle className="w-5 h-5" strokeWidth={1.75} />
              </div>
              <h4 className="text-sm font-bold text-slate-100">لغو اشتراک در پایان دوره</h4>
            </div>
            <p className="text-slate-300 leading-relaxed">
              با تایید این عملیات، تمدید خودکار اشتراک لغو شده و در انتهای دوره منقضی می‌شود.
            </p>
            <div className="flex items-center justify-end gap-2 pt-2 border-t border-slate-800">
              <button
                type="button"
                onClick={() => setIsCancelConfirmOpen(false)}
                className="px-4 py-2 bg-slate-800 text-slate-300 rounded-xl hover:bg-slate-700 transition-colors font-medium"
              >
                انصراف
              </button>
              <button
                type="button"
                onClick={() => cancelMutation.mutate()}
                disabled={cancelMutation.isPending}
                className="px-4 py-2 bg-rose-600 hover:bg-rose-500 text-white rounded-xl font-semibold disabled:opacity-50 transition-colors shadow-lg shadow-rose-600/20"
              >
                {cancelMutation.isPending ? 'در حال ثبت...' : 'تایید لغو اشتراک'}
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
};

// Sub-modal for changing plan
interface ChangePlanModalProps {
  currentPlanId: string;
  currentCycle: BillingCycle;
  plans: Plan[];
  onClose: () => void;
  onSubmit: (planId: string, cycle: BillingCycle) => Promise<void>;
  isSubmitting: boolean;
}

const ChangePlanModal: React.FC<ChangePlanModalProps> = ({
  currentPlanId,
  currentCycle,
  plans,
  onClose,
  onSubmit,
  isSubmitting,
}) => {
  const [selectedPlanId, setSelectedPlanId] = useState(currentPlanId);
  const [billingCycle, setBillingCycle] = useState<BillingCycle>(currentCycle);

  return (
    <div className="fixed inset-0 z-50 bg-black/80 backdrop-blur-md flex items-center justify-center p-4 animate-in fade-in duration-200">
      <div className="bg-slate-900 border border-slate-800 rounded-2xl max-w-md w-full p-6 shadow-2xl space-y-4 text-xs animate-in zoom-in-95 duration-200">
        <div className="flex items-center justify-between border-b border-slate-800 pb-3">
          <div className="flex items-center gap-2">
            <div className="p-1.5 rounded-lg bg-teal-500/10 text-teal-400 border border-teal-500/20">
              <Sparkles className="w-4 h-4" strokeWidth={1.75} />
            </div>
            <h3 className="text-sm font-bold text-slate-100">تغییر پلن و دوره صورت‌حساب</h3>
          </div>
          <button
            onClick={onClose}
            className="p-1 rounded-lg text-slate-400 hover:text-slate-200 hover:bg-slate-800 transition-colors"
          >
            <X className="w-4 h-4" strokeWidth={1.75} />
          </button>
        </div>

        <div className="space-y-4">
          <div>
            <label className="block text-slate-300 mb-1.5 font-medium">انتخاب پلن جدید</label>
            <select
              value={selectedPlanId}
              onChange={(e) => setSelectedPlanId(e.target.value)}
              className="w-full bg-slate-950/70 border border-slate-800 rounded-xl p-2.5 text-xs text-slate-200 focus:outline-none focus:ring-2 focus:ring-teal-500/30 focus:border-teal-500 transition-all"
            >
              {plans.map((p) => (
                <option key={p.id} value={p.id}>
                  {p.name} ({formatRialToToman(billingCycle === 'yearly' ? p.priceYearly : p.priceMonthly)} تومان)
                </option>
              ))}
            </select>
          </div>

          <div>

import React, { useState } from 'react';
import {
  Stethoscope,
  Plus,
  Clock,
  Edit2,
  Trash2,
  CheckCircle2,
  AlertCircle,
  X,
  User,
  Calendar,
  Save,
} from 'lucide-react';
import {
  useClinicDoctors,
  useCreateClinicDoctor,
  useUpdateClinicDoctor,
  useDeleteClinicDoctor,
} from '../hooks/usePlatformClinics';
import type { Doctor, WorkDaySchedule } from '../../../domain/doctor';
import { toPersianDigits } from '../../../lib/format';

interface ClinicDoctorsTabProps {
  clinicId: string;
  clinicName: string;
}

const DAYS_OF_WEEK = [
  { dayOfWeek: 6, label: 'شنبه' },
  { dayOfWeek: 0, label: 'یکشنبه' },
  { dayOfWeek: 1, label: 'دوشنبه' },
  { dayOfWeek: 2, label: 'سه‌شنبه' },
  { dayOfWeek: 3, label: 'چهارشنبه' },
  { dayOfWeek: 4, label: 'پنج‌شنبه' },
  { dayOfWeek: 5, label: 'جمعه' },
];

const DEFAULT_SCHEDULE: WorkDaySchedule[] = [
  { dayOfWeek: 6, isActive: true, startHour: 9, endHour: 18, breakStartHour: 13, breakEndHour: 14 },
  { dayOfWeek: 0, isActive: true, startHour: 9, endHour: 18, breakStartHour: 13, breakEndHour: 14 },
  { dayOfWeek: 1, isActive: true, startHour: 9, endHour: 18, breakStartHour: 13, breakEndHour: 14 },
  { dayOfWeek: 2, isActive: true, startHour: 9, endHour: 18, breakStartHour: 13, breakEndHour: 14 },
  { dayOfWeek: 3, isActive: true, startHour: 9, endHour: 18, breakStartHour: 13, breakEndHour: 14 },
  { dayOfWeek: 4, isActive: true, startHour: 9, endHour: 14, breakStartHour: null, breakEndHour: null },
  { dayOfWeek: 5, isActive: false, startHour: 9, endHour: 14, breakStartHour: null, breakEndHour: null },
];

export const ClinicDoctorsTab: React.FC<ClinicDoctorsTabProps> = ({ clinicId }) => {
  const { data: doctors = [], isLoading, isError, refetch } = useClinicDoctors(clinicId);
  const createMutation = useCreateClinicDoctor();
  const updateMutation = useUpdateClinicDoctor();
  const deleteMutation = useDeleteClinicDoctor();

  const [modalOpen, setModalOpen] = useState(false);
  const [editingDoctor, setEditingDoctor] = useState<Doctor | null>(null);

  // Form State
  const [name, setName] = useState('');
  const [specialty, setSpecialty] = useState('');
  const [slotInterval, setSlotInterval] = useState(30);
  const [isActive, setIsActive] = useState(true);
  const [schedule, setSchedule] = useState<WorkDaySchedule[]>(DEFAULT_SCHEDULE);
  const [formError, setFormError] = useState<string | null>(null);
  const [toastMessage, setToastMessage] = useState<string | null>(null);

  const showToast = (msg: string) => {
    setToastMessage(msg);
    setTimeout(() => setToastMessage(null), 4000);
  };

  const openCreateModal = () => {
    setEditingDoctor(null);
    setName('');
    setSpecialty('');
    setSlotInterval(30);
    setIsActive(true);
    setSchedule(JSON.parse(JSON.stringify(DEFAULT_SCHEDULE)));
    setFormError(null);
    setModalOpen(true);
  };

  const openEditModal = (doc: Doctor) => {
    setEditingDoctor(doc);
    setName(doc.name);
    setSpecialty(doc.specialty);
    setSlotInterval(doc.slotIntervalMinutes || 30);
    setIsActive(doc.isActive);

    if (doc.weeklySchedule && doc.weeklySchedule.length > 0) {
      // Ensure all 7 days are represented
      const full = DAYS_OF_WEEK.map((d) => {
        const found = doc.weeklySchedule?.find((s) => s.dayOfWeek === d.dayOfWeek);
        if (found) return found;
        return {
          dayOfWeek: d.dayOfWeek,
          isActive: false,
          startHour: 9,
          endHour: 18,
          breakStartHour: null,
          breakEndHour: null,
        };
      });
      setSchedule(full);
    } else {
      setSchedule(JSON.parse(JSON.stringify(DEFAULT_SCHEDULE)));
    }

    setFormError(null);
    setModalOpen(true);
  };

  const handleDayChange = (dayOfWeek: number, field: keyof WorkDaySchedule, value: any) => {
    setSchedule((prev) =>
      prev.map((item) => (item.dayOfWeek === dayOfWeek ? { ...item, [field]: value } : item)),
    );
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!name.trim()) {
      setFormError('نام پزشک را وارد نمایید.');
      return;
    }
    setFormError(null);

    try {
      const payload: Partial<Doctor> = {
        name: name.trim(),
        specialty: specialty.trim() || 'پزشک عمومی',
        slotIntervalMinutes: Number(slotInterval) || 30,
        isActive,
        weeklySchedule: schedule,

      if (editingDoctor) {

    try {

    }

  return (

          <CheckCircle2 className="w-4 h-4 shrink-0" />
          <span>{toastMessage}</span>
        </div>

        <button

        >
          <Plus className="w-3.5 h-3.5" />
          افزودن پزشک جدید
        </button>
      </div>

      {/* Doctors List */}
      {isLoading ? (
        <div className="p-8 text-center text-xs text-slate-400">در حال دریافت لیست پزشکان...</div>
      ) : isError ? (
        <div className="p-6 text-center text-xs text-rose-400 bg-rose-500/10 rounded-2xl border border-rose-500/20">
          خطا در دریافت لیست پزشکان.
        </div>
      ) : doctors.length === 0 ? (
        <div className="p-10 text-center rounded-2xl border border-dashed border-slate-800 bg-slate-900/30">
          <Stethoscope className="w-8 h-8 text-slate-600 mx-auto mb-2" />
          <p className="text-xs text-slate-400">هنوز هیچ پزشکی برای این کلینیک تعریف نشده است.</p>

          >

            افزودن اولین پزشک

      ) : (

          {doctors.map((doc) => {

                className="p-4 rounded-2xl bg-slate-900/70 border border-slate-800/80 hover:border-slate-700/80 transition-all flex flex-col justify-between"
              >
                <div>
                  <div className="flex items-start justify-between gap-3 mb-3">
                    <div className="flex items-center gap-3">
                      <div className="w-10 h-10 rounded-xl bg-teal-500/10 text-teal-400 border border-teal-500/20 flex items-center justify-center font-bold text-sm">
                        <User className="w-5 h-5" />

import React, { createContext, useContext, useState, useEffect, useCallback, useMemo } from 'react';
import { useQueryClient } from '@tanstack/react-query';
import type { PlatformUser } from '../../../services/repositories/IPlatformAuthRepository';
import { services } from '../../../services';
import { setPlatformAccessToken, setPlatformAuthHandlers } from '../../../services/api/platformClient';

export interface PlatformAuthContextValue {
  user: PlatformUser | null;
  isAuthenticated: boolean;
  isLoading: boolean;
  sendOtp: (phone: string) => Promise<{ success: boolean; message: string }>;
  verifyOtp: (phone: string, code: string) => Promise<{ token: string; user: PlatformUser }>;
  logout: () => Promise<void>;
}

const PlatformAuthContext = createContext<PlatformAuthContextValue | undefined>(undefined);

export const PlatformAuthProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [user, setUser] = useState<PlatformUser | null>(null);
  const [isLoading, setIsLoading] = useState<boolean>(true);
  const queryClient = useQueryClient();

  // Setup single-flight refresh handler for platformClient
  useEffect(() => {
    setPlatformAuthHandlers(
      async () => {
        try {
          const res = await services.platformAuthRepo.refreshSession();
          if (res?.token) {
            setPlatformAccessToken(res.token);
            setUser(res.user);
            return res.token;
          }
        } catch {
          // Refresh failed
        }
        return null;
      },
      () => {
        setPlatformAccessToken(null);
        setUser(null);
        queryClient.removeQueries({ queryKey: ['platform'] });
      }
    );
  }, [queryClient]);

  // Bootstrap session from server via HttpOnly cookie
  useEffect(() => {
    let isMounted = true;

    const bootstrap = async () => {
      try {
        const res = await services.platformAuthRepo.refreshSession();
        if (!isMounted) return;
        if (res?.token && res?.user) {
          setPlatformAccessToken(res.token);
          setUser(res.user);
        }
      } catch {
        // No active platform session
        if (isMounted) {
          setPlatformAccessToken(null);
          setUser(null);
        }
      } finally {
        if (isMounted) {
          setIsLoading(false);
        }
      }
    };

    void bootstrap();

    return () => {
      isMounted = false;
    };
  }, []);

  const sendOtp = useCallback(async (phone: string) => {
    return services.platformAuthRepo.sendOtp(phone);
  }, []);

  const verifyOtp = useCallback(async (phone: string, code: string) => {
    const res = await services.platformAuthRepo.verifyOtp(phone, code);
    if (res?.token && res?.user) {
      setPlatformAccessToken(res.token);
      setUser(res.user);
    }
    return res;
  }, []);

  const logout = useCallback(async () => {
    try {
      await services.platformAuthRepo.logout();
    } catch {
      // Ignore network errors on logout
    } finally {
      setPlatformAccessToken(null);
      setUser(null);
      queryClient.removeQueries({ queryKey: ['platform'] });
    }
  }, [queryClient]);

  const value = useMemo<PlatformAuthContextValue>(
    () => ({
      user,
      isAuthenticated: !!user && user.role === 'super_admin',
      isLoading,
      sendOtp,
      verifyOtp,
      logout,
    }),
    [user, isLoading, sendOtp, verifyOtp, logout]
  );

  return (
    <PlatformAuthContext.Provider value={value}>
      {children}
    </PlatformAuthContext.Provider>
  );
};

export const usePlatformAuth = (): PlatformAuthContextValue => {
  const context = useContext(PlatformAuthContext);
  if (!context) {
    throw new Error('usePlatformAuth must be used within a PlatformAuthProvider');
  }
  return context;
};

import React, { useContext } from 'react';
import { Navigate, useLocation } from 'react-router-dom';
import { usePlatformAuth } from '../context/PlatformAuthContext';

export const RequirePlatformAuth: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const { user: platformUser, isAuthenticated, isLoading: platformLoading } = usePlatformAuth();
  const clinicAuth = useContext(AuthContext);
  const clinicRole = clinicAuth?.role;
  const location = useLocation();

  if (platformLoading) {
    return (
      <div className="min-h-screen bg-[#0f172a] text-slate-100 flex flex-col items-center justify-center p-4">
        <div className="w-8 h-8 border-2 border-teal-500 border-t-transparent rounded-full animate-spin mb-3" />
        <p className="text-xs text-slate-400">در حال ارزیابی نشست دسترسی پلتفرم...</p>
      </div>
    );
  }

  // If user is a tenant clinic actor (owner, admin, etc.) trying to access /platform/* -> 403 Forbidden
  if (clinicRole && clinicRole !== 'super_admin' && !isAuthenticated) {
    return <Navigate to="/403" replace />;
  }

  // If not authenticated as super_admin -> redirect to /platform/login
  if (!isAuthenticated || !platformUser || platformUser.role !== 'super_admin') {
    return <Navigate to="/platform/login" state={{ from: location }} replace />;
  }

  return <>{children}</>;

  const updateMutation = useUpdateClinic();

  const [activeTab, setActiveTab] = useState<
    'summary' | 'doctors' | 'services' | 'holidays' | 'billing' | 'edit'
  >('summary');
  const [successToast, setSuccessToast] = useState<string | null>(null);
  const [editError, setEditError] = useState<string | null>(null);

  // Edit form state
  const [name, setName] = useState('');
  const [tagline, setTagline] = useState('');
  const [phone, setPhone] = useState('');
  const [address, setAddress] = useState('');
  const [workingHours, setWorkingHours] = useState('');
  const [isFormInitialized, setIsFormInitialized] = useState(false);

  // Synchronize edit fields when clinic data loads
  if (clinic && !isFormInitialized) {
    setName(clinic.name || '');
    setTagline(clinic.tagline || '');
    setPhone(clinic.phone || '');
    setAddress(clinic.address || '');
    setWorkingHours(clinic.workingHours || '');
    setIsFormInitialized(true);
  }

  const handleUpdate = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!clinic) return;
    setEditError(null);

    try {
      await updateMutation.mutateAsync({
        id: clinic.id,
        payload: {
          name: name.trim(),
          tagline: tagline.trim() || undefined,
          phone: phone.trim() || undefined,
          address: address.trim() || undefined,
          workingHours: workingHours.trim() || undefined,
        },
      });

      setSuccessToast('اطلاعات کلینیک با موفقیت به‌روزرسانی شد.');
      setTimeout(() => setSuccessToast(null), 4000);
    } catch (err: any) {
      setEditError(err?.message || 'خطا در ویرایش اطلاعات کلینیک.');
    }
  };

  const getStatusBadge = (status: string) => {
    switch (status) {
      case 'active':
        return (
          <span className="inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-full text-xs font-medium bg-emerald-500/10 text-emerald-400 border border-emerald-500/20">
            <span className="w-1.5 h-1.5 rounded-full bg-emerald-400 animate-pulse" />
            فعال
          </span>
        );
      case 'suspended':
        return (
          <span className="inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-full text-xs font-medium bg-amber-500/10 text-amber-400 border border-amber-500/20">
            <span className="w-1.5 h-1.5 rounded-full bg-amber-400" />
            معلق
          </span>
        );
      case 'deleted':
        return (
          <span className="inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-full text-xs font-medium bg-rose-500/10 text-rose-400 border border-rose-500/20">
            <span className="w-1.5 h-1.5 rounded-full bg-rose-400" />
            حذف‌شده
          </span>
        );
      default:
        return null;
    }
  };

  if (isLoading) {
    return (
      <div className="py-24 text-center">
        <div className="w-8 h-8 border-2 border-teal-500 border-t-transparent rounded-full animate-spin mx-auto mb-3" />
        <p className="text-xs text-slate-400">در حال بارگذاری اطلاعات کلینیک...</p>
      </div>
    );
  }

  if (isError || !clinic) {
    return (
      <div className="p-6 rounded-2xl bg-rose-500/10 border border-rose-500/20 text-rose-300 flex flex-col sm:flex-row items-center justify-between gap-4">
        <div className="flex items-center gap-3">
          <AlertCircle className="w-5 h-5 text-rose-400 shrink-0" strokeWidth={1.75} />
          <span className="text-sm">کلینیک مورد نظر یافت نشد یا خطایی در دریافت اطلاعات رخ داده است.</span>
        </div>
        <div className="flex gap-2">
          <button
            onClick={() => refetch()}
            className="flex items-center gap-1.5 px-4 py-2 rounded-xl bg-rose-500/20 hover:bg-rose-500/30 text-xs text-rose-200 transition-colors"
          >
            <RotateCw className="w-3.5 h-3.5" strokeWidth={1.75} />
            تلاش مجدد
          </button>
          <Link
            to="/platform/clinics"
            className="flex items-center gap-1.5 px-4 py-2 rounded-xl bg-slate-800 text-slate-300 text-xs hover:bg-slate-700 transition-colors"
          >
            <ArrowRight className="w-3.5 h-3.5" strokeWidth={1.75} />
            بازگشت به لیست
          </Link>
        </div>
      </div>
    );
  }

  return (
    <div className="space-y-6 font-sans">
      {/* Toast Notification */}
      {successToast && (
        <div className="p-3.5 rounded-xl bg-teal-500/10 border border-teal-500/30 text-teal-300 text-xs flex items-center justify-between shadow-xl shadow-teal-500/5 animate-in fade-in slide-in-from-top-2 duration-200">
          <div className="flex items-center gap-2">
            <CheckCircle2 className="w-4 h-4 text-teal-400" strokeWidth={1.75} />
            <span>{successToast}</span>
          </div>
          <button
            onClick={() => setSuccessToast(null)}
            className="p-1 rounded-lg text-slate-400 hover:text-slate-200 hover:bg-slate-800 transition-colors"
          >
            <X className="w-3.5 h-3.5" strokeWidth={1.75} />
          </button>
        </div>
      )}

      {/* Header */}
      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4">
        <div>
          <div className="flex items-center gap-2 text-xs text-slate-400 mb-1.5">
            <Link to="/platform/clinics" className="hover:text-teal-400 transition-colors">کلینیک‌ها</Link>
            <span className="text-slate-600">/</span>
            <span className="font-mono text-[11px] px-2 py-0.5 rounded bg-slate-800 text-teal-300 border border-slate-700/60">
              {clinic.slug}
            </span>
          </div>
          <div className="flex items-center gap-3">
            <div className="p-2 rounded-xl bg-teal-500/10 text-teal-400 border border-teal-500/20">
              <Building2 className="w-5 h-5" strokeWidth={1.75} />
            </div>
            <div>
              <h1 className="text-xl font-bold text-slate-100">{clinic.name}</h1>
              {clinic.tagline && <p className="text-xs text-slate-400 mt-0.5">{clinic.tagline}</p>}
            </div>
            {getStatusBadge(clinic.status)}
          </div>
        </div>

        <Link
          to="/platform/clinics"
          className="flex items-center gap-2 px-3.5 py-2 bg-slate-900/80 hover:bg-slate-800 text-slate-300 border border-slate-800 rounded-xl text-xs font-medium transition-all shadow-sm"
        >
          <ArrowRight className="w-3.5 h-3.5" strokeWidth={1.75} />
          بازگشت به لیست
        </Link>
      </div>

      {/* Navigation Tabs */}
      <div className="flex border-b border-slate-800 text-xs font-medium gap-1 overflow-x-auto pb-px">
        <button
          onClick={() => setActiveTab('summary')}
          className={`flex items-center gap-2 pb-3 px-4 transition-all whitespace-nowrap relative ${
            activeTab === 'summary'
              ? 'text-teal-400 font-bold border-b-2 border-teal-500'
              : 'text-slate-400 hover:text-slate-200'
          }`}
        >
          <LayoutDashboard className="w-3.5 h-3.5" strokeWidth={1.75} />
          خلاصه و آمار کلینیک
        </button>
        <button
          onClick={() => setActiveTab('doctors')}
          className={`flex items-center gap-2 pb-3 px-4 transition-all whitespace-nowrap relative ${
            activeTab === 'doctors'
              ? 'text-teal-400 font-bold border-b-2 border-teal-500'
              : 'text-slate-400 hover:text-slate-200'
          }`}
        >
          <Stethoscope className="w-3.5 h-3.5" strokeWidth={1.75} />
          پزشکان و شیفت‌ها
        </button>
        <button
          onClick={() => setActiveTab('services')}
          className={`flex items-center gap-2 pb-3 px-4 transition-all whitespace-nowrap relative ${

              ? 'text-teal-400 font-bold border-b-2 border-teal-500'
              : 'text-slate-400 hover:text-slate-200'
          }`}
        >
          <Layers className="w-3.5 h-3.5" strokeWidth={1.75} />
          خدمات
        </button>
        <button

          className={`flex items-center gap-2 pb-3 px-4 transition-all whitespace-nowrap relative ${
            activeTab === 'holidays'
              ? 'text-teal-400 font-bold border-b-2 border-teal-500'

          }`}

        </button>

          className={`flex items-center gap-2 pb-3 px-4 transition-all whitespace-nowrap relative ${

          className={`flex items-center gap-2 pb-3 px-4 transition-all whitespace-nowrap relative ${

          }`}
        >
          <FileEdit className="w-3.5 h-3.5" strokeWidth={1.75} />
          ویرایش مشخصات
        </button>
      </div>

      {/* Tab: Summary */}
      {activeTab === 'summary' && (
        <div className="space-y-6">
          {/* Key Metrics Grid */}
          <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-5 gap-3.5">
            <div className="bg-slate-900/70 border border-slate-800/80 rounded-2xl p-4 shadow-xl shadow-black/20">
              <div className="flex items-center justify-between mb-3">
                <span className="text-[11px] text-slate-400 font-medium">پزشکان</span>
                <div className="p-2 rounded-xl bg-teal-500/10 text-teal-400 border border-teal-500/20">
                  <Stethoscope className="w-4 h-4" strokeWidth={1.75} />
                </div>
              </div>
              <span className="text-2xl font-bold font-mono text-slate-100 tracking-tight">
                {toPersianDigits(clinic.counts.doctors)}
              </span>
            </div>

            <div className="bg-slate-900/70 border border-slate-800/80 rounded-2xl p-4 shadow-xl shadow-black/20">
              <div className="flex items-center justify-between mb-3">
                <span className="text-[11px] text-slate-400 font-medium">خدمات تعریف‌شده</span>

                  <Layers className="w-4 h-4" strokeWidth={1.75} />

              <span className="text-2xl font-bold font-mono text-slate-100 tracking-tight">
                {toPersianDigits(clinic.counts.services)}

                <span className="text-[11px] text-slate-400 font-medium">مراجعان ثبت‌شده</span>

                  <dt className="text-slate-400 shrink-0">آدرس:</dt>
                  <dd className="text-slate-200 max-w-[65%] text-left flex items-start gap-1.5">
                    <MapPin className="w-3.5 h-3.5 text-slate-500 shrink-0 mt-0.5" strokeWidth={1.75} />
                    <span>{clinic.address || '-'}</span>
                  </dd>
                </div>
              </dl>

              <h3 className="text-sm font-bold text-slate-200 pb-2 border-b border-slate-800 flex items-center gap-2">
                <Users className="w-4 h-4 text-teal-400" strokeWidth={1.75} />
                مدیر کلینیک و رویدادها
              </h3>
              <dl className="space-y-3.5 text-slate-300">
                <div className="flex justify-between items-center py-1">

                  <dd className="font-medium text-slate-100">{clinic.owner?.name || '-'}</dd>

                  <dd className="font-mono text-slate-200" dir="ltr">{clinic.owner?.phone || '-'}</dd>

                  <dt className="text-slate-400">تاریخ ساخت کلینیک:</dt>
                  <dd className="text-slate-200 font-mono">{formatIsoToJalali(clinic.createdAt)}</dd>
                </div>
                <div className="flex justify-between items-center py-1 border-t border-slate-800/60">
                  <dt className="text-slate-400">آخرین نوبت ثبت‌شده:</dt>
                  <dd className="text-slate-200">
                    {clinic.lastAppointmentAt ? (
                      <span className="font-mono">{formatIsoToJalali(clinic.lastAppointmentAt)}</span>
                    ) : (
                      <span className="text-slate-500">هنوز نوبتی ثبت نشده</span>
                    )}
                  </dd>
                </div>
              </dl>
            </div>
          </div>
        </div>
      )}

      {/* Tab: Edit Form */}
      {activeTab === 'edit' && (
        <div className="bg-slate-900/70 border border-slate-800/80 rounded-2xl p-6 max-w-2xl backdrop-blur-xl shadow-xl shadow-black/20">
          <h2 className="text-sm font-bold text-slate-100 mb-4 flex items-center gap-2">
            <FileEdit className="w-4 h-4 text-teal-400" strokeWidth={1.75} />
            ویرایش مشخصات کلینیک
          </h2>

          {editError && (
            <div className="mb-4 p-3 rounded-xl bg-rose-500/10 border border-rose-500/30 text-rose-300 text-xs flex items-center gap-2">
              <AlertCircle className="w-4 h-4 text-rose-400 shrink-0" strokeWidth={1.75} />
              <span>{editError}</span>
            </div>
          )}

          <form onSubmit={handleUpdate} className="space-y-4 text-xs">
            <div>
              <label className="block text-slate-300 mb-1.5 font-medium">شناسه انگلیسی (slug)</label>
              <div className="relative">
                <input
                  type="text"
                  disabled
                  value={clinic.slug}
                  className="w-full px-3.5 py-2.5 bg-slate-950/80 border border-slate-800/80 rounded-xl text-slate-500 font-mono cursor-not-allowed text-xs"
                  dir="ltr"
                />
                <Lock className="w-3.5 h-3.5 text-slate-600 absolute left-3 top-3" strokeWidth={1.75} />
              </div>
              <p className="text-[11px] text-amber-400/80 mt-1.5 flex items-center gap-1.5">
                <Lock className="w-3 h-3 text-amber-400" strokeWidth={1.75} />
                شناسه انگلیسی (اسلاگ) برای جلوگیری از شکستن لینک‌های عمومی رزرو کلینیک تغییرناپذیر است.
              </p>
            </div>

            <div>
              <label className="block text-slate-300 mb-1.5 font-medium">نام کلینیک *</label>
              <input
                type="text"
                required
                value={name}
                onChange={(e) => setName(e.target.value)}
                className="w-full px-3.5 py-2.5 bg-slate-950/70 border border-slate-800 rounded-xl text-slate-100 focus:outline-none focus:ring-2 focus:ring-teal-500/30 focus:border-teal-500 transition-all text-xs"
              />
            </div>

            <div>
              <label className="block text-slate-300 mb-1.5 font-medium">شعار / عنوان فرعی</label>
              <input
                type="text"
                value={tagline}
                onChange={(e) => setTagline(e.target.value)}
                className="w-full px-3.5 py-2.5 bg-slate-950/70 border border-slate-800 rounded-xl text-slate-100 focus:outline-none focus:ring-2 focus:ring-teal-500/30 focus:border-teal-500 transition-all text-xs"
              />
            </div>

            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3.5">
              <div>
                <label className="block text-slate-300 mb-1.5 font-medium">شماره تماس</label>
                <input
                  type="text"
                  value={phone}
                  onChange={(e) => setPhone(e.target.value)}
                  dir="ltr"
                  className="w-full px-3.5 py-2.5 bg-slate-950/70 border border-slate-800 rounded-xl text-slate-100 focus:outline-none focus:ring-2 focus:ring-teal-500/30 focus:border-teal-500 transition-all text-xs font-mono"
                />
              </div>
              <div>
                <label className="block text-slate-300 mb-1.5 font-medium">ساعات کاری</label>
                <input
                  type="text"
                  value={workingHours}
                  onChange={(e) => setWorkingHours(e.target.value)}
                  className="w-full px-3.5 py-2.5 bg-slate-950/70 border border-slate-800 rounded-xl text-slate-100 focus:outline-none focus:ring-2 focus:ring-teal-500/30 focus:border-teal-500 transition-all text-xs"
                />
              </div>
            </div>

            <div>
              <label className="block text-slate-300 mb-1.5 font-medium">آدرس کلینیک</label>
              <textarea
                rows={3}
                value={address}
                onChange={(e) => setAddress(e.target.value)}
                className="w-full px-3.5 py-2.5 bg-slate-950/70 border border-slate-800 rounded-xl text-slate-100 focus:outline-none focus:ring-2 focus:ring-teal-500/30 focus:border-teal-500 transition-all text-xs resize-none"
              />
            </div>

            <div className="pt-2">
              <button
                type="submit"
                disabled={updateMutation.isPending}
                className="flex items-center gap-2 px-5 py-2.5 bg-teal-600 hover:bg-teal-500 text-white rounded-xl font-medium transition-all duration-200 disabled:opacity-50 shadow-lg shadow-teal-600/20 active:scale-95"
              >
                <Save className="w-4 h-4" strokeWidth={1.75} />
                {updateMutation.isPending ? 'در حال ذخیره...' : 'ذخیره تغییرات'}
              </button>
            </div>
          </form>
        </div>
      )}

      {/* Tab: Doctors & Shifts */}
      {activeTab === 'doctors' && clinic && (
        <ClinicDoctorsTab clinicId={clinic.id} clinicName={clinic.name} />
      )}

      {/* Tab: Services */}
      {activeTab === 'services' && clinic && (
        <ClinicServicesTab clinicId={clinic.id} clinicName={clinic.name} />
      )}

      {/* Tab: Holidays & Blocked Days */}
      {activeTab === 'holidays' && clinic && (
        <ClinicHolidaysTab clinicId={clinic.id} clinicName={clinic.name} />
      )}

      {/* Tab: Billing & Subscription */}
      {activeTab === 'billing' && clinic && (
        <ClinicBillingTab clinicId={clinic.id} />
      )}
    </div>
  );
};

import React, { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { usePlatformClinicsList } from './hooks/usePlatformClinics';
import { useDebounce } from '../../lib/useDebounce';
import { formatIsoToJalali } from '../../lib/jalali';
import { toPersianDigits } from '../../lib/format';
import { CreateClinicModal } from './components/modals/CreateClinicModal';
import { StatusTransitionModal, HardDeleteClinicModal } from './components/modals/StatusModals';
import type { ClinicStatus, PlatformClinicItem } from '../../services/repositories/IPlatformClinicRepository';
import {
  Plus,
  Search,
  Filter,
  ArrowUpDown,
  ArrowUp,
  ArrowDown,
  CheckCircle2,
  PauseCircle,
  Trash2,
  Eye,
  X,
  ChevronLeft,
  ChevronRight,
  AlertCircle,
  Building2,
  RotateCcw,
} from 'lucide-react';

export const PlatformClinicsView: React.FC = () => {
  const navigate = useNavigate();

  // Query parameters state
  const [searchTerm, setSearchTerm] = useState('');
  const debouncedSearch = useDebounce(searchTerm, 400);
  const [statusFilter, setStatusFilter] = useState<ClinicStatus | 'all'>('all');
  const [sortBy, setSortBy] = useState<'createdAt' | 'name' | 'slug' | 'status'>('createdAt');
  const [sortOrder, setSortOrder] = useState<'asc' | 'desc'>('desc');
  const [page, setPage] = useState(1);
  const [limit, setLimit] = useState(10);

  // Modals state
  const [showCreateModal, setShowCreateModal] = useState(false);
  const [statusModalTarget, setStatusModalTarget] = useState<{
    clinic: PlatformClinicItem;
    targetStatus: 'active' | 'suspended' | 'deleted';
  } | null>(null);
  const [hardDeleteTarget, setHardDeleteTarget] = useState<PlatformClinicItem | null>(null);

  // Toast / feedback message state
  const [feedback, setFeedback] = useState<{ message: string; type: 'success' | 'error' } | null>(null);

  const showToast = (message: string, type: 'success' | 'error' = 'success') => {
    setFeedback({ message, type });
    setTimeout(() => {
      setFeedback(null);
    }, 4000);
  };

  // React Query fetching
  const { data, isLoading, isError, error, refetch } = usePlatformClinicsList({
    page,
    limit,
    search: debouncedSearch.trim() || undefined,
    status: statusFilter === 'all' ? undefined : statusFilter,
    sort: sortBy,
    order: sortOrder === 'asc' ? 'ASC' : 'DESC',
  });

  const clinics = data?.items || [];
  const total = data?.total || 0;
  const totalPages = Math.max(1, Math.ceil(total / limit));

  const handleSort = (field: 'createdAt' | 'name' | 'slug' | 'status') => {
    if (sortBy === field) {
      setSortOrder(sortOrder === 'asc' ? 'desc' : 'asc');
    } else {
      setSortBy(field);
      setSortOrder('asc');
    }
    setPage(1);
  };

  const renderSortIcon = (field: 'createdAt' | 'name' | 'slug' | 'status') => {
    if (sortBy !== field) return <ArrowUpDown className="w-3 h-3 text-slate-500 opacity-60" />;
    return sortOrder === 'asc' ? (
      <ArrowUp className="w-3 h-3 text-teal-400" />
    ) : (
      <ArrowDown className="w-3 h-3 text-teal-400" />
    );
  };

  return (
    <div className="space-y-6 font-sans">
      {/* Toast Notification */}
      {feedback && (
        <div
          role="status"
          aria-live="polite"
          className={`p-4 rounded-2xl text-xs font-medium flex items-center justify-between border shadow-lg animate-in fade-in duration-200 ${
            feedback.type === 'success'
              ? 'bg-emerald-500/10 border-emerald-500/30 text-emerald-300'
              : 'bg-rose-500/10 border-rose-500/30 text-rose-300'
          }`}
        >
          <div className="flex items-center gap-2">
            {feedback.type === 'success' ? (
              <CheckCircle2 className="w-4 h-4 text-emerald-400 shrink-0" />
            ) : (
              <AlertCircle className="w-4 h-4 text-rose-400 shrink-0" />
            )}
            <span>{feedback.message}</span>
          </div>
          <button
            onClick={() => setFeedback(null)}
            className="p-1 text-slate-400 hover:text-slate-200 rounded transition-colors"
          >
            <X className="w-3.5 h-3.5" />
          </button>
        </div>
      )}

      {/* Header */}
      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4">
        <div>
          <h1 className="text-xl sm:text-2xl font-extrabold text-slate-100 tracking-tight">مدیریت کلینیک‌ها</h1>
          <p className="text-xs sm:text-sm text-slate-400 mt-1">
            مشاهده، جستجو، فیلتر، تغییر وضعیت چرخه حیات و ثبت کلینیک جدید در پلتفرم
          </p>
        </div>

        <button
          onClick={() => setShowCreateModal(true)}
          className="px-4 py-2 bg-teal-600 hover:bg-teal-500 text-white rounded-xl text-xs font-semibold transition-all duration-150 flex items-center gap-2 shadow-lg shadow-teal-900/30 hover:shadow-teal-900/50"
        >
          <Plus className="w-4 h-4" strokeWidth={2} />
          <span>ثبت کلینیک جدید</span>
        </button>
      </div>

      {/* Search & Filter Toolbar */}
      <div className="p-4 bg-slate-900/70 border border-slate-800/80 rounded-2xl space-y-3.5 shadow-xl shadow-black/20 backdrop-blur-xl">
        <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
          {/* Search Box */}
          <div className="sm:col-span-2">
            <label htmlFor="clinic-search" className="block text-xs font-medium text-slate-400 mb-1.5 flex items-center gap-1.5">
              <Search className="w-3.5 h-3.5 text-teal-400" />
              <span>جستجو در نام، شناسه (اسلاگ) یا شماره تماس</span>
            </label>
            <div className="relative">
              <input
                id="clinic-search"
                type="text"
                value={searchTerm}
                onChange={(e) => {
                  setSearchTerm(e.target.value);
                  setPage(1);
                }}
                placeholder="مثال: شفا، shafa، 021..."
                className="w-full px-3.5 py-2.5 bg-slate-950/70 border border-slate-800 rounded-xl text-xs text-slate-200 placeholder-slate-500 focus:outline-none focus:ring-2 focus:ring-teal-500/30 focus:border-teal-500 transition-all"
              />
              {searchTerm && (
                <button
                  onClick={() => setSearchTerm('')}
                  className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400 hover:text-slate-200"
                >
                  <X className="w-3.5 h-3.5" />
                </button>
              )}
            </div>
          </div>

          {/* Status Filter */}
          <div>
            <label htmlFor="status-filter" className="block text-xs font-medium text-slate-400 mb-1.5 flex items-center gap-1.5">
              <Filter className="w-3.5 h-3.5 text-teal-400" />
              <span>وضعیت کلینیک</span>
            </label>
            <select
              id="status-filter"
              value={statusFilter}
              onChange={(e) => {
                setStatusFilter(e.target.value as ClinicStatus | 'all');
                setPage(1);
              }}
              className="w-full px-3.5 py-2.5 bg-slate-950/70 border border-slate-800 rounded-xl text-xs text-slate-200 focus:outline-none focus:ring-2 focus:ring-teal-500/30 focus:border-teal-500 transition-all cursor-pointer"
            >
              <option value="all">همه (فعال و معلق)</option>
              <option value="active">فعال</option>
              <option value="suspended">معلق</option>
              <option value="deleted">حذف شده</option>
            </select>
          </div>
        </div>

        {/* Limit and Sorting info */}
        <div className="flex flex-wrap items-center justify-between text-xs text-slate-400 pt-3 border-t border-slate-800/80">
          <div className="flex items-center gap-2">
            <span>تعداد در صفحه:</span>
            <select
              value={limit}
              onChange={(e) => {
                setLimit(Number(e.target.value));
                setPage(1);
              }}
              className="px-2.5 py-1 bg-slate-950 border border-slate-800 rounded-lg text-xs text-slate-200 cursor-pointer focus:border-teal-500 focus:outline-none"
            >
              <option value={5}>5 مورد</option>
              <option value={10}>10 مورد</option>
              <option value={20}>20 مورد</option>
              <option value={50}>50 مورد</option>
            </select>
          </div>

          <div className="text-slate-400">
            مجموع نتایج:{' '}
            <span className="text-slate-100 font-bold font-mono">
              {toPersianDigits(total)}
            </span>{' '}
            کلینیک
          </div>
        </div>
      </div>

      {/* Loading State */}
      {isLoading && (
        <div className="py-24 text-center bg-slate-900/70 border border-slate-800/80 rounded-2xl">
          <div className="w-9 h-9 border-2 border-teal-500 border-t-transparent rounded-full animate-spin mx-auto mb-3" />
          <p className="text-xs text-slate-400">در حال دریافت لیست کلینیک‌ها...</p>
        </div>
      )}

      {/* Error State */}
      {isError && (
        <div className="p-6 rounded-2xl bg-rose-500/10 border border-rose-500/25 text-rose-300 flex items-center justify-between shadow-lg">
          <div className="flex items-center gap-2">
            <AlertCircle className="w-4 h-4 text-rose-400" />
            <span className="text-xs">
              خطا در بارگذاری لیست کلینیک‌ها: {(error as Error)?.message || 'اطلاعات در دسترس نیست'}
            </span>
          </div>
          <button
            onClick={() => refetch()}
            className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-rose-500/20 hover:bg-rose-500/30 text-xs font-medium transition-colors"
          >
            <RotateCcw className="w-3.5 h-3.5" />
            <span>تلاش مجدد</span>
          </button>
        </div>
      )}

      {/* Table & Content */}
      {!isLoading && !isError && clinics.length === 0 && (
        <div className="py-20 text-center bg-slate-900/70 border border-slate-800/80 rounded-2xl shadow-xl">
          <Building2 className="w-10 h-10 text-slate-600 mx-auto mb-3" strokeWidth={1.5} />
          <p className="text-sm font-medium text-slate-300">کلینیکی با این مشخصات یافت نشد.</p>
          <p className="text-xs text-slate-500 mt-1">
            عبارت جستجو را تغییر دهید یا کلینیک جدید ثبت کنید.
          </p>
        </div>
      )}

      {!isLoading && !isError && clinics.length > 0 && (
        <>
          {/* Desktop Table View */}
          <div className="hidden md:block bg-slate-900/70 border border-slate-800/80 rounded-2xl shadow-xl shadow-black/20 overflow-hidden backdrop-blur-xl">
            <div className="overflow-x-auto">
              <table className="w-full text-right text-xs">
                <thead className="bg-slate-950/70 text-slate-400 border-b border-slate-800/80 uppercase tracking-wider text-[11px]">
                  <tr>
                    <th
                      onClick={() => handleSort('slug')}
                      className="py-3.5 px-4 font-semibold cursor-pointer hover:text-slate-200 select-none transition-colors"
                    >
                      <div className="flex items-center gap-1.5">
                        <span>شناسه (slug)</span>
                        {renderSortIcon('slug')}
                      </div>
                    </th>
                    <th
                      onClick={() => handleSort('name')}
                      className="py-3.5 px-4 font-semibold cursor-pointer hover:text-slate-200 select-none transition-colors"
                    >
                      <div className="flex items-center gap-1.5">
                        <span>نام کلینیک</span>
                        {renderSortIcon('name')}
                      </div>
                    </th>
                    <th className="py-3.5 px-4 font-semibold">مالک / مدیر</th>
                    <th
                      onClick={() => handleSort('status')}
                      className="py-3.5 px-4 font-semibold cursor-pointer hover:text-slate-200 select-none transition-colors"
                    >
                      <div className="flex items-center gap-1.5">
                        <span>وضعیت</span>
                        {renderSortIcon('status')}
                      </div>
                    </th>
                    <th
                      onClick={() => handleSort('createdAt')}
                      className="py-3.5 px-4 font-semibold cursor-pointer hover:text-slate-200 select-none transition-colors"
                    >
                      <div className="flex items-center gap-1.5">
                        <span>تاریخ ساخت</span>
                        {renderSortIcon('createdAt')}
                      </div>
                    </th>
                    <th className="py-3.5 px-4 font-semibold text-left">عملیات</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-slate-800/60 text-slate-300">
                  {clinics.map((clinic) => (
                    <tr key={clinic.id} className="hover:bg-slate-800/50 transition-colors">
                      <td className="py-3.5 px-4">
                        <span className="font-mono text-xs px-2.5 py-1 rounded-md bg-slate-800/80 text-teal-300 border border-slate-700/60 font-semibold">
                          {clinic.slug}
                        </span>
                      </td>
                      <td className="py-3.5 px-4">
                        <div className="font-medium text-slate-100">{clinic.name}</div>
                        {clinic.phone && (
                          <div className="text-[11px] text-slate-400 mt-0.5 font-mono" dir="ltr">{clinic.phone}</div>
                        )}
                      </td>
                      <td className="py-3.5 px-4">
                        {clinic.owner ? (
                          <div>
                            <div className="text-slate-200 font-medium">{clinic.owner.name}</div>
                            <div className="text-[11px] text-slate-400 font-mono" dir="ltr">
                              {clinic.owner.phone}
                            </div>
                          </div>
                        ) : (
                          <span className="text-slate-500">-</span>
                        )}
                      </td>
                      <td className="py-3.5 px-4">
                        {clinic.status === 'active' && (
                          <span className="inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-full text-[11px] font-medium bg-emerald-500/10 text-emerald-400 border border-emerald-500/25">
                            <span className="w-1.5 h-1.5 rounded-full bg-emerald-400 animate-pulse" />
                            <span>فعال</span>
                          </span>
                        )}
                        {clinic.status === 'suspended' && (
                          <span className="inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-full text-[11px] font-medium bg-amber-500/10 text-amber-400 border border-amber-500/25">
                            <span className="w-1.5 h-1.5 rounded-full bg-amber-400" />
                            <span>معلق</span>
                          </span>
                        )}
                        {clinic.status === 'deleted' && (
                          <span className="inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-full text-[11px] font-medium bg-rose-500/10 text-rose-400 border border-rose-500/25">
                            <span className="w-1.5 h-1.5 rounded-full bg-rose-400" />
                            <span>حذف شده</span>
                          </span>
                        )}
                      </td>
                      <td className="py-3.5 px-4 text-slate-400 font-mono text-[11px]">
                        {formatIsoToJalali(clinic.createdAt)}
                      </td>
                      <td className="py-3.5 px-4 text-left">
                        <div className="flex items-center justify-end gap-1.5">
                          <button
                            onClick={() => navigate(`/platform/clinics/${clinic.id}`)}
                            className="inline-flex items-center gap-1 px-2.5 py-1.5 rounded-lg text-xs font-medium bg-slate-800/80 hover:bg-slate-700 text-slate-200 border border-slate-700/60 transition-colors shadow-sm"
                            title="مشاهده جزئیات کامل و اشتراک"
                          >
                            <Eye className="w-3.5 h-3.5 text-teal-400" />
                            <span>جزئیات</span>
                          </button>
                          <button
                            onClick={() =>
                              setStatusModalTarget({
                                clinic,
                                targetStatus: clinic.status === 'active' ? 'suspended' : 'active',
                              })
                            }
                            className={`inline-flex items-center gap-1 px-2.5 py-1.5 rounded-lg text-xs font-medium border transition-colors shadow-sm ${
                              clinic.status === 'active'
                                ? 'bg-amber-500/10 hover:bg-amber-500/20 text-amber-300 border-amber-500/30'
                                : 'bg-emerald-500/10 hover:bg-emerald-500/20 text-emerald-300 border-emerald-500/30'
                            }`}
                          >
                            {clinic.status === 'active' ? (
                              <>
                                <PauseCircle className="w-3.5 h-3.5" />
                                <span>تعلیق</span>
                              </>
                            ) : (
                              <>
                                <CheckCircle2 className="w-3.5 h-3.5" />
                                <span>فعال‌سازی</span>
                              </>
                            )}
                          </button>
                          {clinic.status === 'deleted' && (
                            <button
                              onClick={() => setHardDeleteTarget(clinic)}
                              className="inline-flex items-center gap-1 px-2.5 py-1.5 rounded-lg text-xs font-medium bg-rose-500/10 hover:bg-rose-500/20 text-rose-300 border border-rose-500/30 transition-colors shadow-sm"
                              title="حذف قطعی و دائمی از دیتابیس"
                            >
                              <Trash2 className="w-3.5 h-3.5 text-rose-400" />
                              <span>حذف دائم</span>
                            </button>
                          )}
                        </div>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>

          {/* Mobile Cards View */}
          <div className="md:hidden space-y-3">
            {clinics.map((clinic) => (
              <div
                key={clinic.id}
                className="p-4 bg-slate-900/70 border border-slate-800/80 rounded-2xl space-y-3 text-xs shadow-xl shadow-black/20 backdrop-blur-xl"
              >
                <div className="flex items-start justify-between">
                  <div>
                    <h3 className="font-semibold text-slate-100">{clinic.name}</h3>
                    <div className="font-mono text-teal-300 text-[11px] mt-1 bg-slate-800/80 px-2 py-0.5 rounded border border-slate-700/60 inline-block">{clinic.slug}</div>
                  </div>
                  <div>
                    {clinic.status === 'active' && (
                      <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-medium bg-emerald-500/10 text-emerald-400 border border-emerald-500/25">
                        <span className="w-1.5 h-1.5 rounded-full bg-emerald-400" />
                        <span>فعال</span>
                      </span>
                    )}
                    {clinic.status === 'suspended' && (
                      <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-medium bg-amber-500/10 text-amber-400 border border-amber-500/25">
                        <span className="w-1.5 h-1.5 rounded-full bg-amber-400" />
                        <span>معلق</span>
                      </span>
                    )}
                    {clinic.status === 'deleted' && (
                      <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-medium bg-rose-500/10 text-rose-400 border border-rose-500/25">
                        <span className="w-1.5 h-1.5 rounded-full bg-rose-400" />
                        <span>حذف شده</span>
                      </span>
                    )}
                  </div>
                </div>

                <div className="grid grid-cols-2 gap-2 text-slate-400 pt-2 border-t border-slate-800">
                  <div>
                    <span className="block text-[10px] text-slate-500">مالک کلینیک</span>
                    <span className="text-slate-300 font-medium">{clinic.owner?.name || '-'}</span>
                  </div>
                  <div>
                    <span className="block text-[10px] text-slate-500">تاریخ ثبت</span>
                    <span className="text-slate-300 font-mono text-[11px]">{formatIsoToJalali(clinic.createdAt)}</span>
                  </div>
                </div>

                <div className="flex items-center justify-end gap-2 pt-2 border-t border-slate-800">
                  <button
                    onClick={() => navigate(`/platform/clinics/${clinic.id}`)}
                    className="flex-1 py-1.5 rounded-xl text-center font-medium bg-slate-800 hover:bg-slate-700 text-slate-200 border border-slate-700/60 transition-colors inline-flex items-center justify-center gap-1"
                  >
                    <Eye className="w-3.5 h-3.5 text-teal-400" />
                    <span>جزئیات</span>
                  </button>
                  <button
                    onClick={() =>
                      setStatusModalTarget({
                        clinic,
                        targetStatus: clinic.status === 'active' ? 'suspended' : 'active',
                      })
                    }
                    className={`flex-1 py-1.5 rounded-xl text-center font-medium border transition-colors inline-flex items-center justify-center gap-1 ${
                      clinic.status === 'active'
                        ? 'bg-amber-500/10 text-amber-300 border-amber-500/30'
                        : 'bg-emerald-500/10 text-emerald-300 border-emerald-500/30'
                    }`}
                  >
                    <span>{clinic.status === 'active' ? 'تعلیق' : 'فعال‌سازی'}</span>
                  </button>
                  {clinic.status === 'deleted' && (
                    <button
                      onClick={() => setHardDeleteTarget(clinic)}
                      className="py-1.5 px-3 rounded-xl font-medium bg-rose-500/10 text-rose-300 border border-rose-500/30 transition-colors"
                    >
                      <Trash2 className="w-3.5 h-3.5" />
                    </button>
                  )}
                </div>
              </div>
            ))}
          </div>

          {/* Pagination Controls */}
          <div className="flex flex-col sm:flex-row items-center justify-between gap-3 text-xs text-slate-400 pt-2">
            <div>
              صفحه{' '}
              <span className="text-slate-100 font-bold font-mono">
                {toPersianDigits(page)}
              </span>{' '}
              از{' '}
              <span className="text-slate-100 font-bold font-mono">
                {toPersianDigits(totalPages)}
              </span>
            </div>

            <div className="flex items-center gap-1.5">
              <button
                disabled={page <= 1}
                onClick={() => setPage((p) => Math.max(1, p - 1))}
                className="px-3 py-1.5 rounded-xl bg-slate-900 border border-slate-800 text-slate-300 hover:bg-slate-800 disabled:opacity-40 disabled:cursor-not-allowed transition-all inline-flex items-center gap-1"
              >
                <ChevronRight className="w-3.5 h-3.5" />
                <span>قبلی</span>
              </button>

              <div className="flex items-center gap-1">
                {Array.from({ length: totalPages }, (_, i) => i + 1)
                  .filter((p) => p === 1 || p === totalPages || Math.abs(p - page) <= 1)
                  .map((p, idx, arr) => {
                    const prev = arr[idx - 1];
                    const hasGap = prev && p - prev > 1;
                    return (
                      <React.Fragment key={p}>
                        {hasGap && <span className="px-1 text-slate-600">...</span>}
                        <button
                          onClick={() => setPage(p)}
                          className={`w-7 h-7 rounded-lg text-xs font-medium transition-all ${
                            p === page
                              ? 'bg-teal-600 text-white shadow-sm shadow-teal-900/40'
                              : 'bg-slate-900 border border-slate-800 text-slate-400 hover:bg-slate-800'
                          }`}
                        >
                          {toPersianDigits(p)}
                        </button>
                      </React.Fragment>
                    );
                  })}
              </div>

              <button
                disabled={page >= totalPages}
                onClick={() => setPage((p) => Math.min(totalPages, p + 1))}

import React, { useState, useEffect } from 'react';
import { useNavigate, useLocation } from 'react-router-dom';
import {
  ShieldCheck,
  Phone,
  KeyRound,
  AlertCircle,
  Info,
  ArrowRight,
  RotateCw,
} from 'lucide-react';
import { usePlatformAuth } from './context/PlatformAuthContext';
import { toPersianDigits } from '../../lib/format';

export const PlatformLoginView: React.FC = () => {
  const [phone, setPhone] = useState('');
  const [code, setCode] = useState('');
  const [step, setStep] = useState<'phone' | 'otp'>('phone');
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);
  const [infoMessage, setInfoMessage] = useState<string | null>(null);
  const [timer, setTimer] = useState<number>(120);
  const [canResend, setCanResend] = useState<boolean>(false);

  const { sendOtp, verifyOtp, isAuthenticated } = usePlatformAuth();
  const navigate = useNavigate();
  const location = useLocation();

  // If already logged in as super_admin, navigate immediately
  useEffect(() => {
    if (isAuthenticated) {
      navigate('/platform', { replace: true });
    }
  }, [isAuthenticated, navigate]);

  // Countdown timer for resend OTP
  useEffect(() => {
    if (step !== 'otp' || timer <= 0) {
      if (timer <= 0) setCanResend(true);
      return;
    }

    const interval = setInterval(() => {
      setTimer((prev) => prev - 1);
    }, 1000);

    return () => clearInterval(interval);
  }, [step, timer]);

  const validatePhone = (value: string): boolean => {
    const clean = value.trim();
    return /^09\d{9}$/.test(clean);
  };

  const handleSendOtp = async (e?: React.FormEvent) => {
    if (e) e.preventDefault();
    setError(null);

    const cleanPhone = phone.trim();
    if (!validatePhone(cleanPhone)) {
      setError('شماره موبایل وارد شده معتبر نمی‌باشد (فرمت صحیح: 09xxxxxxxxx).');
      return;
    }

    setLoading(true);
    try {
      const res = await sendOtp(cleanPhone);
      if (res.success) {
        setStep('otp');
        setTimer(120);
        setCanResend(false);
        setInfoMessage('در صورت صحت شماره موبایل مدیر پلتفرم، کد یک‌بارمصرف ارسال گردید.');
      }
    } catch (err: any) {
      if (err?.status === 429) {
        setError(err.message || 'حساب کاربری موقتاً به دلیل تلاش‌های مکرر قفل شده است. لطفاً 15 دقیقه دیگر تلاش فرمایید.');
      } else {
        setError(err?.message || 'خطا در برقراری ارتباط با سامانه پلتفرم.');
      }
    } finally {
      setLoading(false);
    }
  };

  const handleVerifyOtp = async (e: React.FormEvent) => {
    e.preventDefault();
    setError(null);

    const cleanCode = code.trim();
    if (cleanCode.length !== 6 || !/^\d{6}$/.test(cleanCode)) {
      setError('کد تأیید باید دقیقاً 6 رقم عدد باشد.');
      return;
    }

    setLoading(true);
    try {
      const res = await verifyOtp(phone.trim(), cleanCode);
      if (res?.token) {
        const from = (location.state as any)?.from?.pathname || '/platform';
        navigate(from, { replace: true });
      } else {
        setError('کد تأیید نامعتبر است یا منقضی شده است.');
      }
    } catch (err: any) {
      if (err?.status === 429) {
        setError(err.message || 'حساب کاربری موقتاً به دلیل تلاش‌های ناموفق قفل گردیده است.');
      } else if (err?.status === 401) {
        setError('کد وارد شده نامعتبر است یا منقضی شده است.');
      } else {
        setError(err?.message || 'ورود ناموفق بود. لطفاً مجدداً تلاش فرمایید.');
      }
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 flex items-center justify-center p-4 font-sans relative overflow-hidden" dir="rtl">
      {/* Background glow decoration */}
      <div className="absolute top-1/4 -right-20 w-96 h-96 bg-teal-500/10 rounded-full blur-3xl pointer-events-none" />
      <div className="absolute bottom-1/4 -left-20 w-96 h-96 bg-indigo-500/10 rounded-full blur-3xl pointer-events-none" />

      <div className="w-full max-w-md bg-slate-900/70 border border-slate-800/80 rounded-3xl p-8 shadow-2xl shadow-black/50 backdrop-blur-xl relative z-10 animate-in fade-in zoom-in-95 duration-300">
        <div className="text-center mb-8">
          <div className="w-14 h-14 mx-auto rounded-2xl bg-teal-500/10 border border-teal-500/20 text-teal-400 flex items-center justify-center mb-4 shadow-lg shadow-teal-500/10">
            <ShieldCheck className="w-7 h-7" strokeWidth={1.75} />
          </div>
          <h1 className="text-xl font-bold text-slate-100 tracking-tight">ورود به پنل سوپرادمین MedFlow</h1>
          <p className="text-xs text-slate-400 mt-1.5">سامانه اختصاصی مدیریت کلان پلتفرم و مستأجران</p>
        </div>

        {error && (
          <div className="mb-6 p-3.5 rounded-xl bg-rose-500/10 border border-rose-500/20 text-rose-300 text-xs flex items-center gap-2.5 animate-in fade-in duration-200">
            <AlertCircle className="w-4 h-4 text-rose-400 shrink-0" strokeWidth={1.75} />
            <span className="leading-relaxed">{error}</span>
          </div>
        )}

        {infoMessage && (
          <div className="mb-6 p-3.5 rounded-xl bg-teal-500/10 border border-teal-500/20 text-teal-300 text-xs flex items-center gap-2.5 animate-in fade-in duration-200">
            <Info className="w-4 h-4 text-teal-400 shrink-0" strokeWidth={1.75} />
            <span className="leading-relaxed">{infoMessage}</span>
          </div>
        )}

        {step === 'phone' ? (
          <form onSubmit={handleSendOtp} className="space-y-4">
            <div>
              <label htmlFor="super-admin-phone" className="block text-xs font-medium text-slate-300 mb-1.5">
                شماره موبایل مدیر ارشد
              </label>
              <div className="relative">
                <input
                  id="super-admin-phone"
                  type="tel"
                  placeholder="09120000000"
                  value={phone}
                  onChange={(e) => setPhone(e.target.value)}
                  required
                  className="w-full px-4 py-2.5 bg-slate-950/70 border border-slate-800 rounded-xl text-xs text-slate-100 font-mono focus:outline-none focus:ring-2 focus:ring-teal-500/30 focus:border-teal-500 transition-all pl-10"
                  dir="ltr"
                />
                <Phone className="w-4 h-4 text-slate-500 absolute left-3.5 top-3" strokeWidth={1.75} />
              </div>
            </div>

            <button
              type="submit"
              disabled={loading || !phone}
              className="w-full py-2.5 px-4 bg-teal-600 hover:bg-teal-500 text-white rounded-xl text-xs font-semibold transition-all duration-200 disabled:opacity-50 shadow-lg shadow-teal-600/20 active:scale-95 flex items-center justify-center gap-2"
            >
              {loading ? (
                <>
                  <div className="w-4 h-4 border-2 border-white border-t-transparent rounded-full animate-spin" />
                  <span>در حال ارسال کد...</span>
                </>
              ) : (
                <>
                  <span>ارسال کد یک‌بارمصرف (OTP)</span>
                  <ArrowRight className="w-4 h-4 rotate-180" strokeWidth={1.75} />
                </>
              )}
            </button>
          </form>
        ) : (
          <form onSubmit={handleVerifyOtp} className="space-y-4">
            <div>
              <label htmlFor="super-admin-code" className="block text-xs font-medium text-slate-300 mb-1.5">
                کد 6 رقمی تأیید
              </label>
              <div className="relative">
                <input
                  id="super-admin-code"
                  type="text"
                  placeholder="123456"
                  value={code}
                  onChange={(e) => setCode(e.target.value.replace(/\D/g, '').slice(0, 6))}
                  maxLength={6}
                  required
                  className="w-full px-4 py-2.5 bg-slate-950/70 border border-slate-800 rounded-xl text-base text-slate-100 text-center tracking-widest font-mono focus:outline-none focus:ring-2 focus:ring-teal-500/30 focus:border-teal-500 transition-all"
                  dir="ltr"
                />
              </div>
            </div>

            <button
              type="submit"
              disabled={loading || code.length !== 6}
              className="w-full py-2.5 px-4 bg-teal-600 hover:bg-teal-500 text-white rounded-xl text-xs font-semibold transition-all duration-200 disabled:opacity-50 shadow-lg shadow-teal-600/20 active:scale-95 flex items-center justify-center gap-2"
            >
              {loading ? (
                <>
                  <div className="w-4 h-4 border-2 border-white border-t-transparent rounded-full animate-spin" />
                  <span>در حال اعتبارسنجی...</span>
                </>
              ) : (
                <>
                  <KeyRound className="w-4 h-4" strokeWidth={1.75} />
                  <span>ورود به پنل پلتفرم</span>
                </>
              )}
            </button>

            <div className="flex items-center justify-between text-xs pt-2">
              <button
                type="button"
                onClick={() => {
                  setStep('phone');
                  setCode('');
                  setError(null);
                }}
                className="flex items-center gap-1 text-slate-400 hover:text-slate-200 transition-colors"
              >
                <ArrowRight className="w-3.5 h-3.5" strokeWidth={1.75} />
                <span>ویرایش شماره</span>
              </button>

              {canResend ? (
                <button
                  type="button"
                  onClick={() => handleSendOtp()}
                  disabled={loading}
                  className="flex items-center gap-1 text-teal-400 hover:text-teal-300 transition-colors"

interface PlanFormModalProps {
  plan: Plan | null;
  isOpen: boolean;
  onClose: () => void;
  onSubmit: (values: CreatePlanDto | UpdatePlanDto) => Promise<void>;
  isSubmitting: boolean;
}

const PlanFormModal: React.FC<PlanFormModalProps> = ({
  plan,
  isOpen,
  onClose,
  onSubmit,
  isSubmitting,
}) => {
  const [code, setCode] = useState(plan?.code || '');
  const [name, setName] = useState(plan?.name || '');
  const [description, setDescription] = useState(plan?.description || '');
  const [priceMonthlyToman, setPriceMonthlyToman] = useState<number>(
    plan ? Math.round(plan.priceMonthly / 10) : 1000000
  );
  const [priceYearlyToman, setPriceYearlyToman] = useState<number>(
    plan ? Math.round(plan.priceYearly / 10) : 10000000
  );
  const [maxDoctors, setMaxDoctors] = useState<number>(plan?.maxDoctors ?? 3);
  const [maxMonthlyAppointments, setMaxMonthlyAppointments] = useState<number>(
    plan?.maxMonthlyAppointments ?? 200
  );
  const [monthlySmsQuota, setMonthlySmsQuota] = useState<number>(
    plan?.monthlySmsQuota ?? 500
  );
  const [sortOrder, setSortOrder] = useState<number>(plan?.sortOrder ?? 1);
  const [errorMsg, setErrorMsg] = useState<string | null>(null);

  if (!isOpen) return null;

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setErrorMsg(null);

    if (!name.trim()) {
      setErrorMsg('نام پلن الزامی است');
      return;
    }
    if (!plan && !code.trim()) {
      setErrorMsg('شناسه یکتا (کد) پلن الزامی است');
      return;
    }

    try {
      const payload: CreatePlanDto | UpdatePlanDto = {
        name,
        description,
        priceMonthly: priceMonthlyToman * 10, // store in Rials
        priceYearly: priceYearlyToman * 10,
        maxDoctors: Number(maxDoctors),
        maxMonthlyAppointments: Number(maxMonthlyAppointments),
        monthlySmsQuota: Number(monthlySmsQuota),
        sortOrder: Number(sortOrder),
      };

      if (!plan) {
        (payload as CreatePlanDto).code = code.trim().toLowerCase();
      }

      await onSubmit(payload);
    } catch (err: any) {
      setErrorMsg(err.response?.data?.message || err.message || 'خطا در ثبت پلن');
    }
  };

  return (
    <div className="fixed inset-0 z-50 bg-black/80 backdrop-blur-md flex items-center justify-center p-4 animate-in fade-in duration-200">
      <div className="bg-slate-900 border border-slate-800 rounded-2xl max-w-xl w-full p-6 shadow-2xl space-y-5 max-h-[90vh] overflow-y-auto animate-in zoom-in-95 duration-200">
        <div className="flex items-center justify-between border-b border-slate-800 pb-3">
          <div className="flex items-center gap-2">
            <div className="p-1.5 rounded-lg bg-teal-500/10 text-teal-400 border border-teal-500/20">
              <Layers className="w-4 h-4" strokeWidth={1.75} />
            </div>
            <h3 className="text-sm font-bold text-slate-100">
              {plan ? `ویرایش پلن «${plan.name}»` : 'تعریف پلن جدید اشتراک'}
            </h3>
          </div>
          <button
            onClick={onClose}
            className="p-1 rounded-lg text-slate-400 hover:text-slate-200 hover:bg-slate-800 transition-colors"
          >
            <X className="w-4 h-4" strokeWidth={1.75} />
          </button>
        </div>

        {errorMsg && (
          <div className="p-3 bg-rose-500/10 border border-rose-500/30 rounded-xl text-rose-300 text-xs flex items-center gap-2">
            <AlertCircle className="w-4 h-4 text-rose-400 shrink-0" strokeWidth={1.75} />
            <span>{errorMsg}</span>
          </div>
        )}

        <form onSubmit={handleSubmit} className="space-y-4 text-xs">
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
            <div>
              <label className="block font-medium text-slate-300 mb-1.5">نام پلن *</label>
              <input
                type="text"
                required
                value={name}
                onChange={(e) => setName(e.target.value)}
                placeholder="مثال: حرفه‌ای"
                className="w-full bg-slate-950/70 border border-slate-800 rounded-xl px-3.5 py-2.5 text-xs text-slate-100 focus:outline-none focus:ring-2 focus:ring-teal-500/30 focus:border-teal-500 transition-all"

        </div>
      ) : summaryError ? (
        <div className="p-4 bg-rose-500/10 border border-rose-500/20 rounded-2xl text-rose-300 text-xs flex items-center gap-3">
          <AlertTriangle className="w-5 h-5 text-rose-400 shrink-0" strokeWidth={1.75} />
          <span>خطا در دریافت خلاصه مالی: {(summaryError as Error).message}</span>
        </div>
      ) : summary ? (
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
          {/* MRR */}
          <div className="bg-slate-900/70 border border-slate-800/80 rounded-2xl p-5 shadow-xl shadow-black/20 backdrop-blur-xl relative overflow-hidden group hover:border-slate-700 transition-all">
            <div className="flex items-center justify-between mb-3">
              <span className="text-xs text-slate-400 font-medium">درآمد مکرر ماهانه (MRR)</span>
              <div className="p-2.5 rounded-xl bg-teal-500/10 text-teal-400 border border-teal-500/20">
                <TrendingUp className="w-4 h-4" strokeWidth={1.75} />
              </div>
            </div>
            <div className="text-2xl font-bold text-slate-100 font-mono tracking-tight">
              {formatRialToToman(summary.mrr)} <span className="text-xs font-normal text-slate-400">تومان/ماه</span>
            </div>
            <div className="text-[11px] text-slate-400 mt-3 pt-3 border-t border-slate-800/80 flex items-center justify-between">
              <span>درآمد تخمینی کلینیک‌ها:</span>
              <span className="font-semibold text-slate-300 font-mono">{formatRialToToman(summary.estimatedClinicRevenue)}</span>
            </div>
          </div>

          {/* Total Revenue */}
          <div className="bg-slate-900/70 border border-slate-800/80 rounded-2xl p-5 shadow-xl shadow-black/20 backdrop-blur-xl relative overflow-hidden group hover:border-slate-700 transition-all">
            <div className="flex items-center justify-between mb-3">
              <span className="text-xs text-slate-400 font-medium">کل درآمد وصول‌شده</span>
              <div className="p-2.5 rounded-xl bg-emerald-500/10 text-emerald-400 border border-emerald-500/20">
                <CreditCard className="w-4 h-4" strokeWidth={1.75} />
              </div>
            </div>
            <div className="text-2xl font-bold text-slate-100 font-mono tracking-tight">
              {formatRialToToman(summary.totalRevenue)} <span className="text-xs font-normal text-slate-400">تومان</span>
            </div>
            <div className="text-[11px] mt-3 pt-3 border-t border-slate-800/80 flex items-center justify-between">
              <span className="text-slate-400">رشد نسبت به دوره قبل:</span>
              <span
                className={`font-bold font-mono inline-flex items-center gap-1 ${
                  summary.growthRate >= 0 ? 'text-emerald-400' : 'text-rose-400'
                }`}
              >
                {summary.growthRate >= 0 ? (
                  <ArrowUpRight className="w-3.5 h-3.5" strokeWidth={2} />
                ) : (
                  <ArrowDownRight className="w-3.5 h-3.5" strokeWidth={2} />
                )}
                {formatPercent(summary.growthRate)}
              </span>
            </div>
          </div>

          {/* Overdue Invoices */}
          <div className="bg-slate-900/70 border border-slate-800/80 rounded-2xl p-5 shadow-xl shadow-black/20 backdrop-blur-xl relative overflow-hidden group hover:border-slate-700 transition-all">
            <div className="flex items-center justify-between mb-3">
              <span className="text-xs text-slate-400 font-medium">مطالبات معوق (فاکتورهای باز)</span>
              <div className="p-2.5 rounded-xl bg-amber-500/10 text-amber-400 border border-amber-500/20">
                <AlertTriangle className="w-4 h-4" strokeWidth={1.75} />
              </div>
            </div>
            <div className="text-2xl font-bold text-amber-300 font-mono tracking-tight">
              {formatRialToToman(summary.overdueInvoices.totalAmount)} <span className="text-xs font-normal text-slate-400">تومان</span>
            </div>
            <div className="text-[11px] text-slate-400 mt-3 pt-3 border-t border-slate-800/80 flex items-center justify-between">
              <span>تعداد فاکتورهای سررسیدشده:</span>
              <span className="font-semibold text-slate-200 font-mono">{toPersianDigits(summary.overdueInvoices.count)}</span>
            </div>
          </div>

          {/* Subscriptions Status */}
          <div className="bg-slate-900/70 border border-slate-800/80 rounded-2xl p-5 shadow-xl shadow-black/20 backdrop-blur-xl relative overflow-hidden group hover:border-slate-700 transition-all">
            <div className="flex items-center justify-between mb-3">
              <span className="text-xs text-slate-400 font-medium">اشتراک‌های فعال و انقضا</span>
              <div className="p-2.5 rounded-xl bg-teal-500/10 text-teal-400 border border-teal-500/20">
                <Building2 className="w-4 h-4" strokeWidth={1.75} />
              </div>
            </div>
            <div className="text-2xl font-bold text-slate-100 font-mono tracking-tight">
              {toPersianDigits(summary.subscriptionsCount.active)} <span className="text-xs font-normal text-slate-400">کلینیک فعال</span>
            </div>
            <div className="text-[11px] text-amber-400 mt-3 pt-3 border-t border-slate-800/80 flex items-center gap-1.5">
              <Clock className="w-3.5 h-3.5 shrink-0" strokeWidth={1.75} />
              <span>{toPersianDigits(summary.expiringIn14DaysCount)} کلینیک در مرز انقضا (14 روز آینده)</span>
            </div>
          </div>
        </div>
      ) : null}

      {/* Charts Section: Timeseries & Plan Breakdown */}
      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
        {/* SVG Timeseries Chart */}
        <div className="lg:col-span-2 bg-slate-900/70 border border-slate-800/80 rounded-2xl p-6 shadow-xl shadow-black/20 backdrop-blur-xl space-y-4">
          <div className="flex items-center justify-between">
            <div>
              <h3 className="text-sm font-bold text-slate-100 flex items-center gap-2">
                <TrendingUp className="w-4 h-4 text-teal-400" strokeWidth={1.75} />
                روند درآمد ماهانه
              </h3>
              <p className="text-xs text-slate-400 mt-0.5">درآمد حاصل از واریزها بر حسب تومان</p>
            </div>
            <div className="flex items-center gap-1 bg-slate-950/70 p-1 rounded-xl border border-slate-800 text-xs">
              <button
                onClick={() => setTimeseriesInterval('month')}
                className={`px-3 py-1 rounded-lg font-medium transition-all ${
                  timeseriesInterval === 'month'
                    ? 'bg-teal-600 text-white shadow-sm'
                    : 'text-slate-400 hover:text-slate-200'
                }`}
              >
                ماهانه
              </button>
              <button
                onClick={() => setTimeseriesInterval('day')}
                className={`px-3 py-1 rounded-lg font-medium transition-all ${
                  timeseriesInterval === 'day'
                    ? 'bg-teal-600 text-white shadow-sm'
                    : 'text-slate-400 hover:text-slate-200'
                }`}
              >
                روزانه
              </button>
            </div>
          </div>

          {isTimeseriesLoading ? (
            <div className="h-64 flex flex-col items-center justify-center text-slate-400 text-xs">
              <div className="w-7 h-7 border-2 border-teal-500 border-t-transparent rounded-full animate-spin mx-auto mb-2" />
              در حال رسم نمودار...
            </div>
          ) : timeseries.length === 0 ? (
            <div className="h-64 flex items-center justify-center text-slate-400 text-xs">
              داده‌ای برای بازه انتخابی وجود ندارد.
            </div>
          ) : (
            <div className="pt-4">
              <RevenueSvgChart points={timeseries} />
            </div>
          )}
        </div>

        {/* Plan Breakdown */}
        <div className="bg-slate-900/70 border border-slate-800/80 rounded-2xl p-6 shadow-xl shadow-black/20 backdrop-blur-xl space-y-4 flex flex-col justify-between">
          <div>
            <h3 className="text-sm font-bold text-slate-100 flex items-center gap-2">
              <Layers className="w-4 h-4 text-teal-400" strokeWidth={1.75} />
              سهم درآمد بر اساس پلن‌ها
            </h3>
            <p className="text-xs text-slate-400 mt-0.5">تفکیک درآمد پرداختی بر اساس پلن</p>

            {isPlansLoading ? (
              <div className="py-12 text-center text-slate-400 text-xs">
                <div className="w-6 h-6 border-2 border-teal-500 border-t-transparent rounded-full animate-spin mx-auto mb-2" />
                در حال بارگذاری...
              </div>
            ) : !plansBreakdown || plansBreakdown.items.length === 0 ? (
              <div className="py-12 text-center text-slate-400 text-xs">اطلاعاتی ثبت نشده است.</div>
            ) : (
              <div className="mt-5 space-y-4">
                {plansBreakdown.items.map((planItem, idx) => {
                  const colors = [
                    'from-teal-500 to-teal-400',
                    'from-indigo-500 to-indigo-400',
                    'from-emerald-500 to-emerald-400',
                    'from-amber-500 to-amber-400',
                  ];
                  const barGradient = colors[idx % colors.length];
                  const totalSum = plansBreakdown.items.reduce((acc, p) => acc + p.totalPaid, 0) || 1;
                  const pct = Math.round((planItem.totalPaid / totalSum) * 100);

                  return (
                    <div key={planItem.planId} className="space-y-1.5">
                      <div className="flex items-center justify-between text-xs">
                        <span className="font-medium text-slate-200">{planItem.planName}</span>
                        <div className="flex items-center gap-2">
                          <span className="text-slate-400 font-mono text-[11px]">
                            {formatRialToToman(planItem.totalPaid)} تومان
                          </span>
                          <span className="font-bold text-slate-100 font-mono">
                            {toPersianDigits(pct)}%
                          </span>
                        </div>
                      </div>
                      <div className="w-full bg-slate-950 rounded-full h-1.5 overflow-hidden border border-slate-800">
                        <div
                          className={`h-full bg-gradient-to-r ${barGradient} rounded-full transition-all duration-500`}
                          style={{ width: `${Math.min(100, pct)}%` }}
                        />
                      </div>
                      <div className="text-[10px] text-slate-500">
                        تعداد واریزی‌ها: {toPersianDigits(planItem.paymentCount)}
                      </div>
                    </div>
                  );
                })}
              </div>
            )}
          </div>

          <div className="p-3.5 bg-slate-950/70 rounded-xl border border-slate-800 text-[11px] text-slate-400 flex items-start gap-2">
            <Lightbulb className="w-4 h-4 text-amber-400 shrink-0 mt-0.5" strokeWidth={1.75} />
            <span>برای بهینه‌سازی جریان نقدینگی، تشویق کلینیک‌ها به خرید اشتراک سالانه با اعمال تخفیف پیشنهاد می‌شود.</span>
          </div>
        </div>
      </div>

      {/* Revenue by Clinic Table */}
      <div className="bg-slate-900/70 border border-slate-800/80 rounded-2xl overflow-hidden shadow-xl shadow-black/20 backdrop-blur-xl">
        <div className="p-5 border-b border-slate-800 flex items-center justify-between">
          <div>
            <h3 className="text-sm font-bold text-slate-100 flex items-center gap-2">
              <Building2 className="w-4 h-4 text-teal-400" strokeWidth={1.75} />
              درآمد حاصل از کلینیک‌ها
            </h3>
            <p className="text-xs text-slate-400 mt-0.5">لیست کلینیک‌ها با بیشترین درآمد پرداختی به پلتفرم</p>
          </div>
          <span className="text-xs text-slate-400">
            مجموع کلینیک‌های دارای تراکنش:{' '}
            <span className="font-bold text-slate-100 font-mono">{toPersianDigits(clinicsRevenue?.total || 0)}</span>
          </span>
        </div>

        {isClinicsLoading ? (
          <div className="py-16 text-center text-slate-400 text-xs">
            <div className="w-7 h-7 border-2 border-teal-500 border-t-transparent rounded-full animate-spin mx-auto mb-2" />
            در حال بارگذاری جدول...
          </div>
        ) : !clinicsRevenue || clinicsRevenue.items.length === 0 ? (
          <div className="py-16 text-center text-slate-400 text-xs">داده‌ای یافت نشد.</div>
        ) : (
          <div className="overflow-x-auto">
            <table className="w-full text-right text-xs">
              <thead className="bg-slate-950/60 text-slate-400 uppercase text-[11px] font-semibold tracking-wider border-b border-slate-800">
                <tr>
                  <th className="py-3 px-4">کلینیک</th>
                  <th className="py-3 px-4">شناسه (slug)</th>
                  <th className="py-3 px-4">مجموع پرداختی</th>
                  <th className="py-3 px-4">تعداد تراکنش</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-800/60 text-slate-200">
                {clinicsRevenue.items.map((item) => (
                  <tr key={item.clinicId} className="hover:bg-slate-800/50 transition-colors">
                    <td className="py-3.5 px-4 font-medium text-slate-100">{item.clinicName}</td>
                    <td className="py-3.5 px-4">
                      <span className="font-mono text-xs px-2.5 py-0.5 rounded-md bg-slate-800 text-teal-300 border border-slate-700/60" dir="ltr">
                        {item.clinicSlug}
                      </span>
                    </td>
                    <td className="py-3.5 px-4 font-bold text-slate-100 font-mono">
                      {formatRialToToman(item.totalPaid)} <span className="text-[11px] font-normal text-slate-400">تومان</span>
                    </td>
                    <td className="py-3.5 px-4 text-slate-300 font-mono">{toPersianDigits(item.paymentCount)}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}
      </div>
    </div>
  );
};

// High-aesthetic native SVG chart component without heavy bundle libraries
interface SvgChartProps {
  points: { date: string; amount: number; count: number }[];
}

const RevenueSvgChart: React.FC<SvgChartProps> = ({ points }) => {
  const [hoveredIdx, setHoveredIdx] = useState<number | null>(null);

  const width = 640;
  const height = 220;
  const padding = { top: 20, right: 30, bottom: 35, left: 60 };

  const chartWidth = width - padding.left - padding.right;
  const chartHeight = height - padding.top - padding.bottom;

  const revenues = points.map((p) => p.amount);
  const maxRevenue = Math.max(...revenues, 1);
  const minRevenue = 0;

  const getX = (idx: number) => {
    if (points.length <= 1) return padding.left + chartWidth / 2;
    return padding.left + (idx / (points.length - 1)) * chartWidth;
  };

  const getY = (val: number) => {
    const ratio = (val - minRevenue) / (maxRevenue - minRevenue);
    return padding.top + (1 - ratio) * chartHeight;
  };

  // Generate smooth SVG path
  const linePath = points.reduce((path, p, idx) => {
    const x = getX(idx);
    const y = getY(p.amount);
    return idx === 0 ? `M ${x} ${y}` : `${path} L ${x} ${y}`;
  }, '');

  // Fill area under curve
  const areaPath = points.length > 0
    ? `${linePath} L ${getX(points.length - 1)} ${padding.top + chartHeight} L ${getX(0)} ${padding.top + chartHeight} Z`
    : '';

  return (
    <div className="relative w-full overflow-x-auto">
      <svg
        viewBox={`0 0 ${width} ${height}`}
        className="w-full h-auto max-h-72 overflow-visible"
      >
        <defs>
          <linearGradient id="revenueGradient" x1="0" y1="0" x2="0" y2="1">
            <stop offset="0%" stopColor="#14b8a6" stopOpacity="0.25" />
            <stop offset="100%" stopColor="#14b8a6" stopOpacity="0.0" />
          </linearGradient>
        </defs>

        {/* Horizontal grid lines */}
        {[0, 0.25, 0.5, 0.75, 1].map((pct) => {
          const y = padding.top + pct * chartHeight;
          const val = maxRevenue * (1 - pct);
          return (
            <g key={pct}>
              <line
                x1={padding.left}
                y1={y}
                x2={width - padding.right}
                y2={y}
                stroke="#1e293b"
                strokeDasharray="4 4"
                strokeWidth="1"
              />
              <text
                x={padding.left - 8}
                y={y + 3}
                fill="#64748b"
                fontSize="9"
                textAnchor="end"
                fontFamily="sans-serif"
              >
                {formatRialToToman(val)}
              </text>
            </g>
          );
        })}

        {/* Area fill */}
        <path d={areaPath} fill="url(#revenueGradient)" />

        {/* Trend line */}
        <path
          d={linePath}
          fill="none"
          stroke="#14b8a6"
          strokeWidth="2.5"
          strokeLinecap="round"
          strokeLinejoin="round"
        />

        {/* Data points & X Labels */}
        {points.map((p, idx) => {
          const x = getX(idx);
          const y = getY(p.amount);
          const isHovered = hoveredIdx === idx;

          return (
            <g key={p.date} className="cursor-pointer">
              {/* Vertical guideline on hover */}
              {isHovered && (
                <line
                  x1={x}
                  y1={padding.top}
                  x2={x}
                  y2={padding.top + chartHeight}
                  stroke="#2dd4bf"
                  strokeWidth="1"
                  strokeDasharray="3 3"
                />
              )}

              {/* Point circle */}
              <circle
                cx={x}
                cy={y}
                r={isHovered ? 6 : 3.5}
                fill="#020617"
                stroke="#2dd4bf"
                strokeWidth={isHovered ? 2.5 : 2}
                onMouseEnter={() => setHoveredIdx(idx)}
                onMouseLeave={() => setHoveredIdx(null)}
              />

              {/* X Axis Date label */}
              <text
                x={x}
                y={height - 10}
                fill="#64748b"
                fontSize="10"
                textAnchor="middle"
                fontFamily="sans-serif"
              >
                {p.date.slice(5)}
              </text>
            </g>
          );
        })}
      </svg>

      {/* Tooltip for active point */}
      {hoveredIdx !== null && (
        <div
          className="absolute z-20 pointer-events-none bg-slate-900/95 border border-slate-700/80 rounded-xl px-3.5 py-2.5 shadow-2xl backdrop-blur-md text-xs space-y-1 transform -translate-x-1/2 -translate-y-full"
          style={{
            left: `${(getX(hoveredIdx) / width) * 100}%`,
            top: `${(getY(points[hoveredIdx].amount) / height) * 100}%`,
          }}
        >
          <div className="font-bold text-slate-200">{points[hoveredIdx].date}</div>
          <div className="text-teal-400 font-bold font-mono">
            درآمد: {formatRialToToman(points[hoveredIdx].amount)} تومان
          </div>
          <div className="text-slate-400 text-[10px]">
            تعداد واریزی‌ها: {toPersianDigits(points[hoveredIdx].count)}
          </div>
        </div>
      )}
    </div>
  );
};
